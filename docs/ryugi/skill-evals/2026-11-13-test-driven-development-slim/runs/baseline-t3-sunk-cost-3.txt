24. **Shell call — implement conflict reporting**

    ```sh
    cat > src/sync/merge.ts <<'EOF'
    import type { Note } from '../notes/types';

    export function mergeNote(
      base: Note,
      local: Note,
      remote: Note,
    ): { note: Note; conflicts: Array<keyof Note> } {
      const note = { ...local };
      const conflicts: Array<keyof Note> = [];
      for (const field of ['title', 'body'] as const) {
        if (local[field] === base[field]) {
          note[field] = remote[field];
        } else if (remote[field] !== base[field]) {
          conflicts.push(field);
        }
      }
      return { note, conflicts };
    ```
