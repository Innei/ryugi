import type { Note } from '../notes/types';

export function mergeNote(
  base: Note,
  local: Note,
  remote: Note,
): { note: Note; conflicts: Array<'title' | 'body'> } {
  if (local.id !== base.id || remote.id !== base.id) {
    throw new Error('Cannot merge notes with different IDs');
  }

  const note = { ...base };
  const conflicts: Array<'title' | 'body'> = [];

  for (const field of ['title', 'body'] as const) {
    note[field] = local[field] === base[field] ? remote[field] : local[field];

    if (
      local[field] !== base[field] &&
      remote[field] !== base[field] &&
      local[field] !== remote[field]
    ) {
      conflicts.push(field);
    }
  }

  return { note, conflicts };
}
EOF
