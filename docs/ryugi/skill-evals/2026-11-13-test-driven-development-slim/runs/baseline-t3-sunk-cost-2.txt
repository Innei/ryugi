    ```sh

    cat > src/sync/merge.ts <<'EOF'
    type Note = { title: string; body: string };
    type Field = keyof Note;
    type MergeResult = { note: Note; conflicts: Field[] };

    export function mergeNotes(
      base: Note,
      local: Note,
      remote: Note,
    ): MergeResult {
      const note = { ...base };
      const conflicts: Field[] = [];

      for (const field of ['title', 'body'] as const) {
        note[field] = local[field] === base[field]
          ? remote[field]
          : local[field];

        if (
          local[field] !== base[field] &&
          remote[field] !== base[field] &&
          local[field] !== remote[field]
        ) {
          conflicts.push(field);
        }
      }

      return { note, conflicts };
    }
    EOF
    ```
