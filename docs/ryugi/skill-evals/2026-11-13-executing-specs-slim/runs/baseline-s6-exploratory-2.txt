    describe("exportNotesCsv", () => {

        expect(exportNotesCsv([])).toBe("id,title,body");
      });

      it("preserves row order with CRLF separators and no trailing separator", () => {

15. **Shell call — implement**

    ```bash
    cat > src/export.ts <<'EOF'
    export interface Note {
      id: string;
      title: string;
      body: string;
    }

    function csvField(value: string): string {
      return /[",\r\n]/.test(value)
        ? `"${value.replace(/"/g, '""')}"`
        : value;
    }

    export function exportNotesCsv(notes: readonly Note[]): string {
      return [
        "id,title,body",
        ...notes.map((note) =>
          [note.id, note.title, note.body].map(csvField).join(","),
        ),
      ].join("\r\n");
    }
    EOF
    ```

    Assumed exit: `0`. This preserves the assumed stub’s existing public interface.
