   }"""
   new = """function escapeCsvField(value: string): string {
     return /[",\\r\\n]/.test(value)
       ? `"${value.replace(/"/g, '""')}"`

   export function writeRows(notes: readonly ExportNote[]): string {
     return notes
       .map(note =>
         [note.id, note.title, note.updatedAt].map(escapeCsvField).join(',')
