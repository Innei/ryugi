   export function exportNotesCsv(
     notes: readonly { title: string; body: string }[],
   ): string {
     const escape = (value: string): string =>
       /[",\r\n]/.test(value)
         ? `"${value.replace(/"/g, '""')}"`
         : value;

     return [
       'title,body',
       ...notes.map(note => `${escape(note.title)},${escape(note.body)}`),
     ].join('\r\n');
   }
