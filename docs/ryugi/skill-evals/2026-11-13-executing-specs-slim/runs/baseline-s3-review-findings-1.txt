4. **Tool: shell** — add a regression test, using the assumed interfaces discovered above:

```sh
cat > tests/export.streaming.test.ts <<'EOF'
import { expect, test } from 'vitest'
import { exportNotes } from '../src/export'
import { csvHeader } from '../src/csv'

test('streams 200000 notes with destination backpressure', async () => {
  const total = 200_000
  let produced = 0
  let consumed = 0
  let headerWritten = false

  async function* iterate() {
    for (let i = 0; i < total; i++) {
      produced++
      expect(produced - consumed).toBeLessThanOrEqual(1)
      yield {
        id: String(i),
        title: `Note ${i}`,
        body: 'x'.repeat(1024),
        updatedAt: '2026-11-10T00:00:00.000Z',
      }
    }
  }

  const notes = {
    iterate,
    async all() {
      const rows = []
      for await (const note of iterate()) rows.push(note)
      return rows
    },
  }

  const destination = {
    async write(chunk: string) {
      await Promise.resolve()
      if (!headerWritten) {
        expect(chunk).toBe(csvHeader)
        headerWritten = true
        return
      }
      expect(chunk).toContain(`Note ${consumed}`)
      consumed++
    },
  }

  await exportNotes(notes, destination)
  expect(headerWritten).toBe(true)
  expect(consumed).toBe(total)
})
EOF
pnpm test -- tests/export.streaming.test.ts
```

**Assumed full output read:** One test fails at the second generated note: expected outstanding notes `<= 1`, received `2`; stack traces through `notes.all()` and `exportNotes`. Exit 1. This is the expected eager-loading failure, before changing implementation.
