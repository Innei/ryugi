5. **Tool call — implement escaping and run Verify:**

   ```sh
   cat > src/export/writeRows.ts <<'EOF'
   type ExportNote = {
     id: string;
     title: string;
     updatedAt: string;
   };

   function escapeField(value: string): string {
     return /[",\r\n]/.test(value)
       ? `"${value.replace(/"/g, '""')}"`
       : value;
   }

   export function writeRows(notes: readonly ExportNote[]): string {
     return notes
       .map(note => [note.id, note.title, note.updatedAt].map(escapeField).join(','))
       .join('\r\n');
   }
   EOF
   pnpm test
   ```
