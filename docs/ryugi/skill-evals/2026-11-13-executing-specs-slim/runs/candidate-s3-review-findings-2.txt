           produced++;
           yield {

             content: 'Export content',
             updatedAt: '2026-11-10T00:00:00.000Z',
           };
         }
       },
     };

         if (!headerWritten) {

         } else {
           expect(row).toContain(`Note ${written}`);
           written++;

         setImmediate(callback);
       },
     });
