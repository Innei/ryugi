10. Shell call:

    ```sh
    cat > src/export.ts <<'EOF'
    export type ExportNote = {
      title: string;
      body: string;
    };

    function escapeField(value: string): string {
      return /[",\r\n]/.test(value)
        ? `"${value.replace(/"/g, '""')}"`
        : value;
    }

    export function exportNotesCsv(notes: readonly ExportNote[]): string {
      const rows = notes.map(
        (note) => `${escapeField(note.title)},${escapeField(note.body)}`,
      );
      return ['title,body', ...rows].join('\r\n') + '\r\n';
    }
    EOF
    pnpm test
    ```
