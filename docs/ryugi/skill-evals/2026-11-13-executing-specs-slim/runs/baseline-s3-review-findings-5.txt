   it('exports 200k notes with bounded read-ahead', async () => {
     let produced = 0;
     let consumed = 0;
     let headerSeen = false;
     let pending = '';
     let peakReadAhead = 0;

     async function* notes(): AsyncGenerator<Note> {
       for (let index = 0; index < 200_000; index++) {
         peakReadAhead = Math.max(peakReadAhead, produced - consumed);
         if (produced - consumed >= 64) {
           throw new Error('export read ahead by 64 notes');
         }

         produced++;
         yield {
           id: String(index),
           title: `Note ${index}`,
           body: 'x'.repeat(8192),
           updatedAt: '2026-11-10T00:00:00.000Z',
         };
       }
     }

     const destination = new Writable({
       highWaterMark: 1024,
       write(chunk, _encoding, callback) {
         pending += chunk.toString();

         let newline: number;
         while ((newline = pending.indexOf('\n')) !== -1) {
           pending = pending.slice(newline + 1);
           if (headerSeen) consumed++;
           else headerSeen = true;
         }

         setImmediate(callback);
       },
     });

     await exportNotes(notes(), destination);

     expect(produced).toBe(200_000);
     expect(consumed).toBe(200_000);
     expect(pending).toBe('');
     expect(peakReadAhead).toBeLessThan(64);
   });
   ```

   This bounds the regression test itself: the old implementation fails after 64 notes rather than exhausting the Mac’s memory. The destination discards consumed data.

   export async function exportNotes(
     notes: AsyncIterable<Note>,
     destination: Writable,
   ): Promise<void> {
     async function* rows(): AsyncGenerator<string> {
       yield csvHeader;
       for await (const note of notes) {
         yield encodeNote(note);
       }
     }

     await pipeline(
       Readable.from(rows(), { objectMode: false }),
       destination,
     );
   }
   ```

   **Assumed existing helper contract:** `csvHeader` and `encodeNote` already include the required row terminators. This preserves their formatting and error behavior while eliminating whole-account accumulation.
