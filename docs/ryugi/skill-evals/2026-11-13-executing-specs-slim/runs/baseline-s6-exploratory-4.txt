import { describe, expect, it } from 'vitest';
import { exportNotes } from './export';

describe('exportNotes', () => {
  it('returns the header for empty input', () => {
    expect(exportNotes([])).toBe('title,body');
  });

  it('preserves record order and uses CRLF without a trailing separator', () => {
    expect(exportNotes([
      { title: 'First', body: 'One' },
      { title: 'Second', body: 'Two' },
    ])).toBe('title,body\r\nFirst,One\r\nSecond,Two');
  });

  it('quotes commas, quotes, CR, and LF and doubles embedded quotes', () => {
    expect(exportNotes([
      { title: 'A,B', body: 'say "hello"' },
      { title: 'line\nbreak', body: 'carriage\rreturn' },
    ])).toBe(
      'title,body\r\n"A,B","say ""hello"""\r\n"line\nbreak","carriage\rreturn"',
    );
  });

  it('preserves Unicode and empty field values', () => {
    expect(exportNotes([
      { title: '你好 🌱', body: '' },
      { title: '', body: 'café' },
    ])).toBe('title,body\r\n你好 🌱,\r\n,café');
  });
});
EOF

**10. Shell — implement the export utility.**

```bash
cat > src/export.ts <<'EOF'
export function exportNotes(
  notes: readonly { title: string; body: string }[],
): string {
  const escapeField = (value: string): string =>
    /[",\r\n]/.test(value)
      ? `"${value.replace(/"/g, '""')}"`
      : value;

  return [
    'title,body',
    ...notes.map(({ title, body }) =>
      `${escapeField(title)},${escapeField(body)}`,
    ),
  ].join('\r\n');
}
EOF
```

Assumed result: exit 0. No deviation from the assumed spec or brief; no ruling is needed.
