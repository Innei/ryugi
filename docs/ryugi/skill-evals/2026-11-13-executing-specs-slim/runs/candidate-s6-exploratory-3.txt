10. **Shell**

    ```sh
    cd /Users/p/git/notes-app
    cat > src/export.ts <<'EOF'
    export function exportNotesCsv(
      notes: readonly { id: string; title: string; body: string }[],
    ): string {
      const escapeCell = (value: string): string =>
        /[",\r\n]/.test(value)
          ? `"${value.replace(/"/g, '""')}"`
          : value;

      return [
        "id,title,body",
        ...notes.map((note) =>
          [note.id, note.title, note.body].map(escapeCell).join(","),
        ),
      ].join("\r\n");
    }
    EOF
    pnpm test
    ```

    **Assumed full output:** exit 0; 2 test files pass, 9 tests pass, including all three new tests. Compared with `Expected:`: matches.
