**8. Implement and verify**

```sh
cat > src/export.ts <<'EOF'
type ExportNote = {
  title: string
  body: string
}

function csvField(value: string): string {
  return /[",\r\n]/.test(value)
    ? `"${value.replace(/"/g, '""')}"`
    : value
}

export function exportNotesCsv(notes: readonly ExportNote[]): string {
  return [
    'title,body',
    ...notes.map(note =>
      [csvField(note.title), csvField(note.body)].join(','),
    ),
  ].join('\r\n')
}
EOF
pnpm test
```
