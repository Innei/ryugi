```ts
function escapeCsvField(value: string): string {
  return /[",\r\n]/.test(value)
    ? `"${value.replace(/"/g, '""')}"`
    : value;
}

export function serializeRows(notes: readonly ExportNote[]): string {
  return notes
    .map((note) =>
      [note.id, note.title, note.updatedAt]
        .map(escapeCsvField)
        .join(','),
    )
    .join('\n');
}
```

Assume the existing `ExportNote` type has string fields `id`, `title`, and `updatedAt`.
