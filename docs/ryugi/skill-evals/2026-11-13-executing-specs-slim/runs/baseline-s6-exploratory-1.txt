12. **Shell — implement**
    ```sh
    cat > src/export.ts <<'EOF'
    export function exportNotesCsv(
      notes: readonly { title: string; body: string }[],
    ): string {
      const quote = (value: string): string =>
        `"${value.replace(/"/g, '""')}"`

      return [
        'title,body',
        ...notes.map(note => `${quote(note.title)},${quote(note.body)}`),
      ].join('\r\n') + '\r\n'
    }
    ```
