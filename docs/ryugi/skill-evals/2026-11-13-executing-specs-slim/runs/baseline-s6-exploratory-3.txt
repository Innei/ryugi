10. **Shell call** — implement the specified behavior and run Verify.

    ```bash
    cat > src/export.ts <<'EOF'
    export function exportNotesCsv(
      notes: readonly { title: string; body: string }[],
    ): string {
      const escapeField = (value: string): string =>
        /[",\r\n]/.test(value)
          ? `"${value.replace(/"/g, '""')}"`
          : value

      return [
        'title,body',
        ...notes.map(
          note => `${escapeField(note.title)},${escapeField(note.body)}`,
        ),
      ].join('\r\n')
    }
    EOF
    pnpm test
    ```
