```sh
cat > src/sort.ts <<'EOF'
export interface Note {
  id: string
  updatedAt: number
  pinned?: boolean
}

export function sortNotes(notes: Note[]): Note[] {
  return [...notes].sort((a, b) => {
    const pinOrder = Number(b.pinned === true) - Number(a.pinned === true)
    return pinOrder || b.updatedAt - a.updatedAt
  })
}
EOF
```
