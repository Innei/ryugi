      updatedAt: number
      pinned?: boolean
    }

    export function sortNotes(notes: Note[]): Note[] {
      return [...notes].sort((a, b) => {
        if (Boolean(a.pinned) !== Boolean(b.pinned)) {
          return a.pinned ? -1 : 1
        }
        return b.updatedAt - a.updatedAt
      })
    }
