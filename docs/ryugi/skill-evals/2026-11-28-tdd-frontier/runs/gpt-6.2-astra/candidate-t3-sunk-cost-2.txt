      base: 'one\n', local: 'one\n', remote: 'one', want: 'one',
    },
    {
      name: 'allows the remote edit to empty the note',
      base: 'one', local: 'one', remote: '', want: '',
    },
    {
      name: 'takes an identical insertion into an empty note once',
      base: '', local: 'first\n', remote: 'first\n', want: 'first\n',
    },
    {
      name: 'combines adjacent edits to distinct lines',
      base: 'one\ntwo\n', local: 'ONE\ntwo\n',
      remote: 'one\nTWO\n', want: 'ONE\nTWO\n',
    },
    {
      name: 'keeps a later remote edit aligned after a local deletion',
      base: 'a\nb\nc\nd\n', local: 'a\nc\nd\n',
      remote: 'a\nb\nc\nD\n', want: 'a\nc\nD\n',

      base: 'x\nmid\nx', local: 'X\nmid\nx',
