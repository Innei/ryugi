### 2. Add tests against the public function

**Shell** — this is the complete new test file.

```sh
cat > src/sync/merge.test.ts <<'EOF'
import { describe, expect, it } from 'vitest'
import { mergeNotes } from './merge'

const base = { id: 'n1', title: 'Plan', body: 'Draft' }

describe('mergeNotes', () => {
  it.each([
    {
      name: 'unchanged copies',
      local: {},
      remote: {},
      wantTitle: 'Plan',
      wantBody: 'Draft',
    },
    {
      name: 'a local-only edit',
      local: { title: 'Today' },
      remote: {},
      wantTitle: 'Today',
      wantBody: 'Draft',
    },
    {
      name: 'a remote-only edit',
      local: {},
      remote: { body: 'Published' },
      wantTitle: 'Plan',
      wantBody: 'Published',
    },
    {
      name: 'a local rename and remote body clear',
      local: { title: 'Today' },
      remote: { body: '' },
      wantTitle: 'Today',
      wantBody: '',
    },
    {
      name: 'a remote rename and local body clear',
      local: { body: '' },
      remote: { title: 'Today' },
      wantTitle: 'Today',
      wantBody: '',
    },
    {
      name: 'identical concurrent edits',
      local: { title: 'Today', body: '' },
      remote: { title: 'Today', body: '' },
      wantTitle: 'Today',
      wantBody: '',
    },
  ])('merges $name', ({ local, remote, wantTitle, wantBody }) => {
    const result = mergeNotes(
      Object.freeze({ ...base }),
      Object.freeze({ ...base, ...local }),
      Object.freeze({ ...base, ...remote }),
    )

    expect(result).toEqual({
      kind: 'merged',
      note: { id: 'n1', title: wantTitle, body: wantBody },
    })
  })

  it.each([
    {
      name: 'title',
      local: { title: 'Today', body: 'Shared' },
      remote: { title: 'Tomorrow', body: 'Shared' },
      fields: ['title'],
    },
    {
      name: 'body',
      local: { body: '' },
      remote: { body: 'Published' },
      fields: ['body'],
    },
    {
      name: 'title and body',
      local: { title: 'Today', body: 'First' },
      remote: { title: 'Tomorrow', body: 'Second' },
      fields: ['title', 'body'],
    },
  ])('reports conflicting fields for $name', ({ local, remote, fields }) => {
    const result = mergeNotes(
      Object.freeze({ ...base }),
      Object.freeze({ ...base, ...local }),
      Object.freeze({ ...base, ...remote }),
    )

    expect(result).toEqual({ kind: 'conflict', fields })
  })
})
EOF
pnpm test src/sync/merge.test.ts
```

```sh

if pnpm test src/sync/merge.test.ts; then
  exit 1
fi
```
