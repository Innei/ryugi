```ts
export type SortDirection = 'asc' | 'desc';

export function sortBy<T extends Record<string, unknown>>(
  items: readonly T[],
  key: keyof T,
  direction: SortDirection = 'asc',
): T[] {
  return [...items].sort((a, b) => {
    const aValue = a[key];
    const bValue = b[key];
    if (aValue < bValue) return direction === 'asc' ? -1 : 1;
    if (aValue > bValue) return direction === 'asc' ? 1 : -1;
    return 0;
  });
}
```
