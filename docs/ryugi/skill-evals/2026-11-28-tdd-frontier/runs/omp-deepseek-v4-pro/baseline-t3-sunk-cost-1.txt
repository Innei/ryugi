```ts
export interface Note {
  id: string;
  title: string;
  content: string;
  tags: string[];
  updatedAt: number;
}

export type MergeField = 'title' | 'content' | 'tags';

export interface MergeResult {
  note: Note;
  conflicts: MergeField[];
}

const fields: MergeField[] = ['title', 'content', 'tags'];

const equal = (a: unknown, b: unknown): boolean =>
  Array.isArray(a) && Array.isArray(b)
    ? a.length === b.length && a.every((v, i) => v === b[i])
    : a === b;

export function mergeNotes(base: Note, ours: Note, theirs: Note): MergeResult {
  if (ours.id !== base.id || theirs.id !== base.id) {
    throw new Error('cannot merge notes with different ids');
  }

  const note: Note = { ...base };
  const conflicts: MergeField[] = [];

  for (const field of fields) {
    const oursChanged = !equal(ours[field], base[field]);
    const theirsChanged = !equal(theirs[field], base[field]);

    if (equal(ours[field], theirs[field])) {
      note[field] = ours[field];
    } else if (oursChanged && theirsChanged) {
      conflicts.push(field);
      note[field] = ours[field];
    } else if (oursChanged) {
      note[field] = ours[field];
    } else if (theirsChanged) {
      note[field] = theirs[field];
    } else {
      note[field] = base[field];
    }
  }

  note.updatedAt = Math.max(ours.updatedAt, theirs.updatedAt);

  return { note, conflicts };
}
```
