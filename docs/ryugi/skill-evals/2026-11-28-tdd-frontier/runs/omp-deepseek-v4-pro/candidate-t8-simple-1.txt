=== sort.ts ===
export function sortBy<T>(items: readonly T[], key: keyof T): T[] {
  return [...items].sort((a, b) => {
    const av = a[key];
    const bv = b[key];
    if (typeof av === 'string' && typeof bv === 'string') {
      return av.localeCompare(bv);
    }
    return Number(av) - Number(bv);
  });
}
