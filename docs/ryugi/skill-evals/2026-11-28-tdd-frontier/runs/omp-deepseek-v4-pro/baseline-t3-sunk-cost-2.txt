import { describe, expect, it } from 'vitest';
import type { Note } from '../note';
import { threeWayMerge } from './merge';

const note = (over: Partial<Note> = {}): Note => ({
  id: 'n1',
  title: 'Title',
  body: 'one\ntwo\nthree',
  ...over,
});

describe('threeWayMerge', () => {
  it('returns the note unchanged when base, mine, and theirs are identical', () => {
    const base = note();
    const result = threeWayMerge(base, base, base);
    expect(result.conflicts).toEqual([]);
    expect(result.note).toEqual(base);
  });

  it('adopts mine when only mine changed', () => {
    const base = note();
    const mine = note({ title: 'Mine title', body: 'one\nTWO\nthree' });
    const result = threeWayMerge(base, mine, base);
    expect(result.conflicts).toEqual([]);
    expect(result.note).toEqual(mine);
  });

  it('adopts theirs when only theirs changed', () => {
    const base = note();
    const theirs = note({ body: 'one\ntwo\nTHREE' });
    const result = threeWayMerge(base, base, theirs);
    expect(result.conflicts).toEqual([]);
    expect(result.note).toEqual(theirs);
  });

  it('stays clean when both sides made the same change', () => {
    const base = note();
    const changed = note({ title: 'Renamed' });
    const result = threeWayMerge(base, changed, changed);
    expect(result.conflicts).toEqual([]);
    expect(result.note).toEqual(changed);
  });

  it('combines changes to different fields without conflict', () => {
    const base = note();
    const mine = note({ title: 'Mine title' });
    const theirs = note({ body: 'one\ntwo\nTHREE' });
    const result = threeWayMerge(base, mine, theirs);
    expect(result.conflicts).toEqual([]);
    expect(result.note).toEqual({ ...base, title: 'Mine title', body: 'one\ntwo\nTHREE' });
  });

  it('reports a conflict and keeps mine when both sides rename the title differently', () => {
    const base = note();
    const result = threeWayMerge(
      base,
      note({ title: 'Mine title' }),
      note({ title: 'Theirs title' }),
    );
    expect(result.conflicts).toEqual(['title']);
    expect(result.note.title).toBe('Mine title');
    expect(result.note.body).toBe(base.body);
  });

  it('merges non-overlapping body edits from both sides', () => {
    const base = note();
    const mine = note({ body: 'ONE\ntwo\nthree' });
    const theirs = note({ body: 'one\ntwo\nTHREE' });
    const result = threeWayMerge(base, mine, theirs);
    expect(result.conflicts).toEqual([]);
    expect(result.note.body).toBe('ONE\ntwo\nTHREE');
  });

  it('stays clean when both sides make the same body edit', () => {
    const base = note();
    const edited = note({ body: 'one\nTWO\nthree' });
    const result = threeWayMerge(base, edited, edited);
    expect(result.conflicts).toEqual([]);
    expect(result.note.body).toBe('one\nTWO\nthree');

import type { Note } from '../note';

export type ConflictField = 'title' | 'body';

export interface MergeResult {
  note: Note;
  conflicts: ConflictField[];
}

interface Hunk {
  baseStart: number;
  baseEnd: number;
  replacement: string[];
}

export function threeWayMerge(base: Note, mine: Note, theirs: Note): MergeResult {
  const conflicts: ConflictField[] = [];

  const title = mergeScalar(base.title, mine.title, theirs.title);
  if (title.conflict) conflicts.push('title');

  const body = mergeLines(base.body.split('\n'), mine.body.split('\n'), theirs.body.split('\n'));
  if (body.conflict) conflicts.push('body');

  return {
    note: { id: base.id, title: title.value, body: body.lines.join('\n') },
    conflicts,
  };
}

function mergeScalar(base: string, mine: string, theirs: string): { value: string; conflict: boolean } {
  if (mine === theirs) return { value: mine, conflict: false };
  if (theirs === base) return { value: mine, conflict: false };
  if (mine === base) return { value: theirs, conflict: false };
  return { value: mine, conflict: true };
}

function mergeLines(base: string[], mine: string[], theirs: string[]): { lines: string[]; conflict: boolean } {
  const mineHunks = hunks(base, mine);
  const theirsHunks = hunks(base, theirs);

  const lines: string[] = [];
  let conflict = false;
  let pos = 0;
  let mi = 0;
  let ti = 0;

  const emitBase = (end: number) => {
    for (let i = pos; i < Math.min(end, base.length); i++) lines.push(base[i]);
    pos = Math.max(pos, Math.min(end, base.length));
  };

  while (mi < mineHunks.length || ti < theirsHunks.length) {
    const mineHunk = mi < mineHunks.length ? mineHunks[mi] : null;
    const theirsHunk = ti < theirsHunks.length ? theirsHunks[ti] : null;

    if (mineHunk && mineHunk.baseEnd <= pos) { mi++; continue; }
    if (theirsHunk && theirsHunk.baseEnd <= pos) { ti++; continue; }

    if (mineHunk && theirsHunk && mineHunk.baseEnd <= theirsHunk.baseStart) {
      emitBase(mineHunk.baseStart);
      lines.push(...mineHunk.replacement);
      pos = mineHunk.baseEnd;
      mi++;
    } else if (mineHunk && theirsHunk && theirsHunk.baseEnd <= mineHunk.baseStart) {
      emitBase(theirsHunk.baseStart);
      lines.push(...theirsHunk.replacement);
      pos = theirsHunk.baseEnd;
      ti++;
    } else if (mineHunk && theirsHunk) {
      const start = Math.min(mineHunk.baseStart, theirsHunk.baseStart);
      const end = Math.max(mineHunk.baseEnd, theirsHunk.baseEnd);
      emitBase(start);
      if (arraysEqual(mineHunk.replacement, theirsHunk.replacement)) {
        lines.push(...mineHunk.replacement);
      } else {
        conflict = true;
        lines.push(
          '<<<<<<< mine',
          ...mineHunk.replacement,
          '=======',
          ...theirsHunk.replacement,
          '>>>>>>> theirs',
        );
      }
      pos = end;
      mi++;
      ti++;
    } else if (mineHunk) {
      emitBase(mineHunk.baseStart);
      lines.push(...mineHunk.replacement);
      pos = mineHunk.baseEnd;
      mi++;
    } else if (theirsHunk) {
      emitBase(theirsHunk.baseStart);
      lines.push(...theirsHunk.replacement);
      pos = theirsHunk.baseEnd;
      ti++;
    }
  }

  emitBase(base.length);
  return { lines, conflict };
}

function hunks(base: string[], changed: string[]): Hunk[] {
  const n = base.length;
  const m = changed.length;
  const lcs: number[][] = Array.from({ length: n + 1 }, () => new Array(m + 1).fill(0));
  for (let i = n - 1; i >= 0; i--) {
    for (let j = m - 1; j >= 0; j--) {
      lcs[i][j] = base[i] === changed[j]
        ? lcs[i + 1][j + 1] + 1
        : Math.max(lcs[i + 1][j], lcs[i][j + 1]);
    }
  }

  const result: Hunk[] = [];
  let i = 0;
  let j = 0;
  let hunkStart = -1;
  const replacement: string[] = [];

  const flush = () => {
    if (hunkStart >= 0) {
      result.push({ baseStart: hunkStart, baseEnd: i, replacement: [...replacement] });
      hunkStart = -1;
      replacement.length = 0;
    }
  };

  while (i < n || j < m) {
    if (i < n && j < m && base[i] === changed[j]) {
      flush();
      i++;
      j++;
    } else if (i < n && (j >= m || lcs[i + 1][j] >= lcs[i][j + 1])) {
      if (hunkStart < 0) hunkStart = i;
      i++;
    } else {
      if (hunkStart < 0) hunkStart = i;
      replacement.push(changed[j]);
      j++;
    }
  }
  flush();

  return result;
}

function arraysEqual(a: string[], b: string[]): boolean {
  return a.length === b.length && a.every((v, i) => v === b[i]);
}
EOF
