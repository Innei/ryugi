5. **Simulated implementation edit**

   The heredoc contains the complete replacement file.

   ```sh
   cat > src/sort.ts <<'EOF'
   export type Note = {
     id: string
     updatedAt: number
     pinned?: boolean
   }

   export function sortNotes(notes: readonly Note[]): Note[] {
     return [...notes].sort((a, b) => {
       const pinOrder = Number(b.pinned ?? false) - Number(a.pinned ?? false)
       return pinOrder || b.updatedAt - a.updatedAt
     })
   }
   EOF
   ```
