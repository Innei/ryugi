import { describe, expect, it } from 'vitest';
import { mergeNotes } from './merge';

const base = {
  id: 'n1',
  title: 'Draft',
  body: 'alpha\nbeta\ngamma',
};

describe('mergeNotes', () => {
  it('preserves an unchanged note', () => {
    expect(mergeNotes(base, { ...base }, { ...base })).toEqual({
      kind: 'merged',
      note: { id: 'n1', title: 'Draft', body: 'alpha\nbeta\ngamma' },
    });
  });

  it('takes a local-only body edit', () => {
    const local = { ...base, body: 'ALPHA\nbeta\ngamma' };

    expect(mergeNotes(base, local, base)).toEqual({
      kind: 'merged',
      note: { id: 'n1', title: 'Draft', body: 'ALPHA\nbeta\ngamma' },
    });
  });

  it('takes a remote-only body edit', () => {
    const remote = { ...base, body: 'alpha\nbeta\nGAMMA' };

    expect(mergeNotes(base, base, remote)).toEqual({
      kind: 'merged',
      note: { id: 'n1', title: 'Draft', body: 'alpha\nbeta\nGAMMA' },
    });
  });

  it('keeps an identical insertion once', () => {
    const body = 'alpha\ninserted\nbeta\ngamma';

    expect(mergeNotes(base, { ...base, body }, { ...base, body })).toEqual({
      kind: 'merged',
      note: { id: 'n1', title: 'Draft', body: 'alpha\ninserted\nbeta\ngamma' },
    });
  });

  it('combines a title edit with an empty body', () => {
    const local = { ...base, title: 'Renamed' };
    const remote = { ...base, body: '' };

    expect(mergeNotes(base, local, remote)).toEqual({
      kind: 'merged',
      note: { id: 'n1', title: 'Renamed', body: '' },
    });
  });

  it('combines separated body edits in document order', () => {
    const local = { ...base, body: 'alpha\nbeta\nGAMMA' };
    const remote = { ...base, body: 'ALPHA\nbeta\ngamma' };

    expect(mergeNotes(base, local, remote)).toEqual({
      kind: 'merged',
      note: { id: 'n1', title: 'Draft', body: 'ALPHA\nbeta\nGAMMA' },
    });
  });

  it('reports conflicting replacements of the same line', () => {
    const local = { ...base, body: 'alpha\nlocal\ngamma' };
    const remote = { ...base, body: 'alpha\nremote\ngamma' };

    expect(mergeNotes(base, local, remote)).toEqual({
      kind: 'conflict',
      fields: ['body'],
    });
  });

  it('reports different insertions at the same position', () => {
    const local = { ...base, body: 'alpha\nlocal\nbeta\ngamma' };
    const remote = { ...base, body: 'alpha\nremote\nbeta\ngamma' };

    expect(mergeNotes(base, local, remote)).toEqual({
      kind: 'conflict',
      fields: ['body'],
    });
  });

  it('reports divergent title edits', () => {
    const local = { ...base, title: 'Local title' };
    const remote = { ...base, title: 'Remote title' };

    expect(mergeNotes(base, local, remote)).toEqual({
      kind: 'conflict',
      fields: ['title'],
    });
  });

  it('combines adjacent body replacements', () => {
    const local = { ...base, body: 'ALPHA\nbeta\ngamma' };
    const remote = { ...base, body: 'alpha\nBETA\ngamma' };

    expect(mergeNotes(base, local, remote)).toEqual({
      kind: 'merged',
      note: { id: 'n1', title: 'Draft', body: 'ALPHA\nBETA\ngamma' },
    });
  });
});
