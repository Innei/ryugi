import { cleanup, render, screen, within } from '@testing-library/react'
import { afterEach, describe, expect, it } from 'vitest'
import { PresenterWindow } from './PresenterWindow'

afterEach(cleanup)

describe('PresenterWindow', () => {
  it('places the current and next slides in separate labeled regions', () => {
    render(
      <PresenterWindow
        currentSlide={<div>Current slide content</div>}
        nextSlide={<div>Next slide content</div>}
        notes=""
      />,
    )

    expect(
      within(screen.getByRole('region', { name: 'Current slide' }))
        .getByText('Current slide content'),
    ).toBeTruthy()
    expect(
      within(screen.getByRole('region', { name: 'Next slide' }))
        .getByText('Next slide content'),
    ).toBeTruthy()
  })

  it('shows speaker notes in their own labeled region', () => {
    render(
      <PresenterWindow
        currentSlide={<div>Current slide content</div>}
        nextSlide={null}
        notes={'Opening point\nClosing point'}
      />,
    )

    expect(
      within(screen.getByRole('region', { name: 'Speaker notes' }))
        .getByText(/Opening point\s+Closing point/),
    ).toBeTruthy()
  })

  it('shows an end-of-deck state when there is no next slide', () => {
    render(
      <PresenterWindow
        currentSlide={<div>Last slide content</div>}
        nextSlide={null}
        notes=""
      />,
    )

    expect(
      within(screen.getByRole('region', { name: 'Next slide' }))
        .getByText('End of deck'),
    ).toBeTruthy()
    expect(screen.getByText('Last slide content')).toBeTruthy()
  })
})
EOF

9. 写入最小布局实现，完整文件内容如下。

```sh
cat > src/presenter/PresenterWindow.tsx <<'EOF'
import type { ReactNode } from 'react'

type PresenterWindowProps = {
  currentSlide: ReactNode
  nextSlide: ReactNode | null
  notes: string
}

export function PresenterWindow({
  currentSlide,
  nextSlide,
  notes,
}: PresenterWindowProps) {
  return (
    <main
      aria-label="Presenter window"
      style={{
        display: 'grid',
        gridTemplateColumns: 'minmax(0, 2fr) minmax(0, 1fr)',
        gridTemplateRows: 'minmax(0, 1fr) minmax(0, 1fr)',
        gap: 16,
        height: '100dvh',
        boxSizing: 'border-box',
        padding: 16,
      }}
    >
      <section
        aria-label="Current slide"
        style={{
          gridRow: '1 / -1',
          minWidth: 0,
          minHeight: 0,
          overflow: 'auto',
        }}
      >
        {currentSlide}
      </section>
      <section
        aria-label="Next slide"
        style={{ minWidth: 0, minHeight: 0, overflow: 'auto' }}
      >
        {nextSlide === null ? <p>End of deck</p> : nextSlide}
      </section>
      <section
        aria-label="Speaker notes"
        style={{
          minWidth: 0,
          minHeight: 0,
          overflow: 'auto',
          whiteSpace: 'pre-wrap',
          overflowWrap: 'anywhere',
        }}
      >
        {notes}
      </section>
    </main>
  )
}
EOF
```
