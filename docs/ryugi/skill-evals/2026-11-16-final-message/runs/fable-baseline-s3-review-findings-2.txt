**Tool call 4 — edit `src/export.ts`** (replace the body at line 40 onward)
```ts
export const EXPORT_PAGE_SIZE = 1000

export async function* exportNotes(userId: string, repo: NotesRepo): AsyncGenerator<string> {
  yield HEADER + '\r\n'
  let after: string | undefined
  for (;;) {
    const page = await repo.findPageByUser(userId, { after, limit: EXPORT_PAGE_SIZE })
    if (page.length === 0) return
    yield page.map(toCsvRow).join('\r\n') + '\r\n'
    if (page.length < EXPORT_PAGE_SIZE) return
    after = page[page.length - 1].id
  }
}
```
