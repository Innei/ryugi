+import type { Slide } from '../model/slide';
+import { SlidePreview } from './SlidePreview';
+import './PresenterView.css';
+
+interface PresenterViewProps {
+  slides: readonly Slide[];
+  currentIndex: number;
+}
+
+export function PresenterView({
+  slides,
+  currentIndex,
+}: PresenterViewProps) {
+  const currentSlide = slides[currentIndex];
+  const nextSlide = slides[currentIndex + 1];
+
+  return (
+    <main className="presenter-view">
+      <section
+        className="presenter-view__current"
+        aria-label="Current slide"
+      >
+        <h2>Current slide</h2>
+        {currentSlide ? (
+          <SlidePreview slide={currentSlide} />
+        ) : (
+          <p>No current slide</p>
+        )}
+      </section>
+      <section className="presenter-view__next" aria-label="Next slide">
+        <h2>Next slide</h2>
+        {nextSlide ? (
+          <SlidePreview slide={nextSlide} />
+        ) : (
+          <p>No next slide</p>
+        )}
+      </section>
+      <section className="presenter-view__notes" aria-label="Speaker notes">
+        <h2>Speaker notes</h2>
+        <div className="presenter-view__notes-content">
+          {currentSlide?.speakerNotes}
+        </div>
+      </section>
+    </main>
+  );
+}
