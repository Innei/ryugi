```ts
const PAGE_SIZE = 500

export async function exportNotes(userId: string, store: NoteStore, out: Writable): Promise<void> {
  await write(out, HEADER + '\r\n')
  let after: string | undefined
  for (;;) {
    const page = await store.listNotesPage(userId, { after, limit: PAGE_SIZE })
    if (page.length === 0) break
    await write(out, page.map(toCsvRow).join('\r\n') + '\r\n')
    after = page[page.length - 1].id
  }
}

// Wait for drain when the destination is slower than the database.
function write(out: Writable, chunk: string): Promise<void> {
  return out.write(chunk) ? Promise.resolve() : once(out, 'drain').then(() => {})
}
```
