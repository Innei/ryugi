import { describe, expect, it } from 'vitest';
import { exportNotesCsv } from './export';

describe('exportNotesCsv', () => {
  it('returns a header for empty input', () => {
    expect(exportNotesCsv([])).toBe('title,body\r\n');
  });

  it('preserves record order and terminates every record with CRLF', () => {
    expect(exportNotesCsv([
      { title: 'Second', body: 'B' },
      { title: 'First', body: 'A' },
    ])).toBe('title,body\r\nSecond,B\r\nFirst,A\r\n');
  });

  it('quotes commas and doubles embedded quotes', () => {
    expect(exportNotesCsv([
      { title: 'a,b', body: 'say "hi"' },
    ])).toBe('title,body\r\n"a,b","say ""hi"""\r\n');
  });

  it('quotes fields containing either carriage returns or line feeds', () => {
    expect(exportNotesCsv([
      { title: 'a\rb', body: 'c\nd' },
      { title: 'e\r\nf', body: 'plain' },
    ])).toBe('title,body\r\n"a\rb","c\nd"\r\n"e\r\nf",plain\r\n');
  });

  it('preserves Unicode and empty fields', () => {
    expect(exportNotesCsv([
      { title: '笔记 🌱', body: '' },
      { title: '', body: '内容' },
    ])).toBe('title,body\r\n笔记 🌱,\r\n,内容\r\n');
  });

  it('accepts frozen notes without modifying them', () => {
    const note = Object.freeze({ title: 'Title', body: 'Body' });
    const notes = Object.freeze([note]);

    expect(exportNotesCsv(notes)).toBe('title,body\r\nTitle,Body\r\n');
    expect(notes).toEqual([{ title: 'Title', body: 'Body' }]);
  });
});
EOF
```

function csvField(value: string): string {
  return /[",\r\n]/.test(value)
    ? `"${value.replace(/"/g, '""')}"`
    : value;
}

export function exportNotesCsv(
  notes: readonly { readonly title: string; readonly body: string }[],
): string {
  return [
    'title,body',
    ...notes.map((note) => `${csvField(note.title)},${csvField(note.body)}`),
  ].join('\r\n') + '\r\n';
}
EOF
```
