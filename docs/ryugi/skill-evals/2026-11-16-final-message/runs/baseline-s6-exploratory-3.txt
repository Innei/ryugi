8. **Shell call — implementation**

   ```sh
   cat > src/export.ts <<'EOF'
   function csvField(value: string): string {
     return /[",\r\n]/.test(value)
       ? `"${value.replace(/"/g, '""')}"`
       : value;
   }

   export function notesToCsv(
     notes: readonly { title: string; body: string }[],
   ): string {
     return [
       'title,body',
       ...notes.map(({ title, body }) => `${csvField(title)},${csvField(body)}`),
     ].join('\r\n') + '\r\n';
   }
   EOF
   pnpm test
   ```
