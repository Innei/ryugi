```tsx
import { SlideCanvas } from '../viewer/SlideCanvas'
import { useDeckStore } from '../store/useDeckStore'

export function PresenterWindow({ deckId }: { deckId: string }) {
  const deck = useDeckStore((s) => s.decks[deckId])
  const currentIndex = useDeckStore((s) => s.currentIndex[deckId] ?? 0)
  if (!deck) return null
  const current = deck.slides[currentIndex]
  const next = deck.slides[currentIndex + 1]
  const notes = current?.notes?.trim()

  return (
    <div
      style={{
        display: 'grid',
        gridTemplateColumns: '2fr 1fr',
        gridTemplateRows: 'auto 1fr',
        gap: 16,
        height: '100vh',
        padding: 16,
      }}
    >
      <section aria-label="Current slide" style={{ gridRow: '1 / span 2' }}>
        <SlideCanvas deck={deck} index={currentIndex} />
      </section>
      <section aria-label="Next slide">
        {next ? <SlideCanvas deck={deck} index={currentIndex + 1} /> : <p>End of deck</p>}
      </section>
      <section aria-label="Speaker notes" style={{ overflowY: 'auto' }}>
        {notes ? <p style={{ whiteSpace: 'pre-wrap' }}>{notes}</p> : <p>No notes</p>}
      </section>
    </div>
  )
}
```
