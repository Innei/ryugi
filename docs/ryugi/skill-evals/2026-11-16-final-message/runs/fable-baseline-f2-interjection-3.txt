```tsx
import type { ReactNode } from 'react'
import type { Slide } from '../model/slide'
import { SlideThumbnail } from '../viewer/SlideThumbnail'

export interface PresenterLayoutProps {
  slides: Slide[]
  currentIndex: number
  timerSlot: ReactNode
}

export function PresenterLayout({ slides, currentIndex, timerSlot }: PresenterLayoutProps) {
  const current = slides[currentIndex]
  const next = slides[currentIndex + 1]
  return (
    <div className="presenter-layout">
      <section aria-label="Current slide" className="presenter-stage">
        {current && <SlideThumbnail slide={current} />}
      </section>
      <aside className="presenter-side">
        <section aria-label="Next slide" className="presenter-next">
          {next ? <SlideThumbnail slide={next} /> : <p className="presenter-end">End of deck</p>}
        </section>
        <div className="presenter-timer">{timerSlot}</div>
        <section aria-label="Speaker notes" className="presenter-notes">
          {current?.notes}
        </section>
      </aside>
    </div>
  )
}
```
