```tsx
import type { ReactNode } from 'react'
import type { Slide } from '../deck/types'
import { SlideView } from '../viewer/SlideView'

export interface PresenterLayoutProps {
  slides: Slide[]
  currentIndex: number
  timerSlot?: ReactNode
}

export function PresenterLayout({ slides, currentIndex, timerSlot }: PresenterLayoutProps) {
  const current = slides[currentIndex]
  const next = slides[currentIndex + 1]

  return (
    <div
      style={{
        display: 'grid',
        gridTemplateColumns: '2fr 1fr',
        gridTemplateRows: 'auto 1fr 1fr',
        gridTemplateAreas: '"header header" "current next" "current notes"',
        gap: 16,
        height: '100vh',
        padding: 16,
      }}
    >
      <header style={{ gridArea: 'header' }}>{timerSlot}</header>
      <section aria-label="Current slide" style={{ gridArea: 'current' }}>
        {current && <SlideView slide={current} />}
      </section>
      <section aria-label="Next slide" style={{ gridArea: 'next' }}>
        {next ? <SlideView slide={next} /> : <p>End of deck</p>}
      </section>
      <section aria-label="Speaker notes" style={{ gridArea: 'notes', overflowY: 'auto' }}>
        {current?.notes}
      </section>
    </div>
  )
}
```
