```ts
import { describe, it, expect } from 'vitest'
import { exportNotes } from '../src/export'

const note = (i: number) => ({ id: `n${i}`, title: `t${i}`, body: 'b', createdAt: new Date(0), updatedAt: new Date(0) })

function pagedStore(total: number) {
  const calls: Array<{ after?: string; limit: number }> = []
  return {
    calls,
    listAll: async () => { throw new Error('listAll must not be used by export') },
    listNotes: async (_u: string, o: { after?: string; limit: number }) => {
      calls.push(o)
      const start = o.after ? Number(o.after.slice(1)) + 1 : 0
      return Array.from({ length: Math.max(0, Math.min(o.limit, total - start)) }, (_, k) => note(start + k))
    },
  }
}

describe('exportNotes streaming', () => {
  it('fetches pages lazily instead of loading every note', async () => {
    const store = pagedStore(2500)
    const it = exportNotes('u1', store as any, { pageSize: 500 })[Symbol.asyncIterator]()
    await it.next() // header
    await it.next() // first page of rows
    expect(store.calls.length).toBe(1)
    expect(store.calls[0].limit).toBe(500)
  })

  it('emits every note exactly once across pages', async () => {
    const store = pagedStore(1201)
    let out = ''
    for await (const chunk of exportNotes('u1', store as any, { pageSize: 500 })) out += chunk
    const lines = out.trimEnd().split('\r\n')
    expect(lines.length).toBe(1202)
    expect(lines[1].startsWith('n0,')).toBe(true)
    expect(lines[1201].startsWith('n1200,')).toBe(true)
  })
})
```

**Call 4 (edit `src/export.ts`):** replace the load-all body with a generator.
```ts
export async function* exportNotes(
  userId: string,
  store: NoteStore,
  { pageSize = 500 }: { pageSize?: number } = {},
): AsyncGenerator<string> {
  yield HEADER + '\r\n'
  let after: string | undefined
  for (;;) {
    const page = await store.listNotes(userId, { after, limit: pageSize })
    if (page.length === 0) return
    yield page.map(toCsvRow).join('\r\n') + '\r\n'
    after = page[page.length - 1].id
  }
}
```
