**Call 11 — write** `src/export.ts`:
```ts
import type { Note } from './types'

const HEADER = ['id', 'title', 'body', 'createdAt'] as const

function escapeField(value: string): string {
  return /[",\r\n]/.test(value) ? `"${value.replace(/"/g, '""')}"` : value
}

export function exportNotesToCsv(notes: Note[]): string {
  const rows = [
    HEADER.join(','),
    ...notes.map((n) => HEADER.map((key) => escapeField(String(n[key]))).join(',')),
  ]
  return rows.map((row) => row + '\r\n').join('')
}
```
