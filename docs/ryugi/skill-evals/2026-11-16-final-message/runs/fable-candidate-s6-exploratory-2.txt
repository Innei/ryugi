```ts
const HEADER = "id,title,body,created_at\r\n";

describe("exportNotesCsv", () => {
  it("returns only the header for an empty list", () => {
    expect(exportNotesCsv([])).toBe(HEADER);
  });

  it("writes one CRLF-terminated row per note, in input order", () => {
    const csv = exportNotesCsv([
      { id: "2", title: "Second", body: "b", createdAt: "2026-11-02T00:00:00Z" },
      { id: "1", title: "First", body: "a", createdAt: "2026-11-01T00:00:00Z" },
    ]);
    expect(csv).toBe(
      HEADER +
        "2,Second,b,2026-11-02T00:00:00Z\r\n" +

    expect(csv).toBe(
      HEADER + '1,"Milk, eggs","say ""hi""\nbye",2026-11-01T00:00:00Z\r\n',
```

```ts
import type { Note } from "./types";

const HEADER = ["id", "title", "body", "created_at"];

function field(value: string): string {
  return /[",\r\n]/.test(value) ? `"${value.replace(/"/g, '""')}"` : value;
}

export function exportNotesCsv(notes: Note[]): string {
  const rows = [
    HEADER,
    ...notes.map((n) => [n.id, n.title, n.body, n.createdAt]),
  ];
  return rows.map((r) => r.map(field).join(",") + "\r\n").join("");
}
```
