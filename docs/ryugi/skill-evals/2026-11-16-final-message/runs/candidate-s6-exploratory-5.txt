+type ExportableNote = {
+  id: string;
+  title: string;
+  body: string;
+};
+
+function escapeCsvField(value: string): string {
+  return /[",\r\n]/.test(value)
+    ? `"${value.replace(/"/g, '""')}"`
+    : value;
+}
+
+export function exportNotesCsv(notes: readonly ExportableNote[]): string {
+  const rows = [
+    'id,title,body',
+    ...notes.map((note) =>
+      [note.id, note.title, note.body].map(escapeCsvField).join(','),
+    ),
+  ];
+  return `${rows.join('\r\n')}\r\n`;
+}
*** End Patch
