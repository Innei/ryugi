async function collect(chunks: AsyncIterable<string>): Promise<string> {
  let out = ''
  for await (const c of chunks) out += c
  return out
}

function pagedFakeStore(total: number) {
  const limits: number[] = []
  const notes = Array.from({ length: total }, (_, i) => ({
    id: String(i + 1).padStart(8, '0'),
    title: `t${i + 1}`,
    body: 'b',
    createdAt: '2026-01-01T00:00:00Z',
    updatedAt: '2026-01-02T00:00:00Z',
  }))
  return {
    limits,
    async listNotes(_userId: string, opts?: { after?: string; limit?: number }) {
      if (!opts?.limit) throw new Error('unbounded listNotes call')
      limits.push(opts.limit)
      const start = opts.after ? notes.findIndex((n) => n.id === opts.after) + 1 : 0
      return notes.slice(start, start + opts.limit)
    },
  }
}

describe('exportNotes on large accounts', () => {
  it('reads notes in bounded pages and still exports every note', async () => {
    const store = pagedFakeStore(2500)
    const csv = await collect(exportNotes('u1', store))
    expect(Math.max(...store.limits)).toBeLessThanOrEqual(500)
    expect(csv.trimEnd().split('\n')).toHaveLength(2501) // header + 2500 rows
    expect(csv).toContain('00002500,t2500')
  })

  it('fetches the next page only when the consumer asks for more', async () => {
    const store = pagedFakeStore(2500)
    const it = exportNotes('u1', store)[Symbol.asyncIterator]()
    await it.next() // header
    await it.next() // first page of rows
    expect(store.limits).toHaveLength(1)
  })
})

```ts
const PAGE_SIZE = 500

export async function* exportNotes(userId: string, store: NoteStore): AsyncGenerator<string> {
  yield HEADER + '\n'
  let after: string | undefined
  for (;;) {
    const page = await store.listNotes(userId, { after, limit: PAGE_SIZE })
    if (page.length === 0) return
    yield page.map(toCsvRow).join('\n') + '\n'
    if (page.length < PAGE_SIZE) return
    after = page[page.length - 1].id
  }
}
```
