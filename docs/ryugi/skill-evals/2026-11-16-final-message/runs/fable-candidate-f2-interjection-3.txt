```tsx
import type { ReactNode } from 'react'
import type { Deck } from '../deck/types'
import { SlideCanvas } from '../viewer/SlideCanvas'

type Props = { deck: Deck; currentIndex: number; timer: ReactNode }

export function PresenterLayout({ deck, currentIndex, timer }: Props) {
  const current = deck.slides[currentIndex]
  const next = deck.slides[currentIndex + 1]
  return (
    <div
      style={{
        display: 'grid',
        gridTemplateColumns: '2fr 1fr',
        gridTemplateRows: 'auto 1fr auto',
        gridTemplateAreas: '"current next" "current notes" "timer notes"',
        gap: 16,
        height: '100%',
      }}
    >
      <section aria-label="Current slide" style={{ gridArea: 'current' }}>
        <SlideCanvas deck={deck} slide={current} />
      </section>
      <section aria-label="Next slide" style={{ gridArea: 'next' }}>
        {next ? <SlideCanvas deck={deck} slide={next} /> : <p>End of deck</p>}
      </section>
      <section aria-label="Speaker notes" style={{ gridArea: 'notes', overflow: 'auto' }}>
        {current.notes}
      </section>
      <section aria-label="Timer" style={{ gridArea: 'timer' }}>
        {timer}
      </section>
    </div>
  )
}
```
