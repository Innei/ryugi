   export interface Note {
     title: string;
     content: string;
     createdAt: string;
   }

   function escapeField(value: string): string {
     return /[",\r\n]/.test(value)
       ? `"${value.replace(/"/g, '""')}"`
       : value;
   }

   export function exportNotesCsv(notes: readonly Note[]): string {
     const rows = ['title,content,createdAt'];

     for (const note of notes) {
       rows.push(
         [note.title, note.content, note.createdAt].map(escapeField).join(','),
       );
     }

     return `${rows.join('\r\n')}\r\n`;
   }
   EOF
   pnpm test
