```ts
export const EXPORT_PAGE_SIZE = 500

export async function* exportNotes(userId: string, store: NoteStore): AsyncGenerator<string> {
  yield CSV_HEADER + '\n'
  let afterId: string | undefined
  for (;;) {
    const page = await store.list(userId, { limit: EXPORT_PAGE_SIZE, afterId })
    if (page.length > 0) yield page.map(toCsvRow).join('\n') + '\n'
    if (page.length < EXPORT_PAGE_SIZE) return
    afterId = page[page.length - 1].id
  }
}
```
