```sh
cat > src/export.ts <<'EOF'
export function exportNotesCsv(
  notes: readonly { id: string; title: string; body: string }[],
): string {
  const escape = (value: string): string =>
    /[",\r\n]/.test(value)
      ? `"${value.replace(/"/g, '""')}"`
      : value;

  const rows = notes.map((note) =>
    [note.id, note.title, note.body].map(escape).join(','),
  );

  return ['id,title,body', ...rows].join('\r\n') + '\r\n';
}
EOF
pnpm test
```
