```ts
const EXPORT_PAGE_SIZE = 500

export async function* exportNotes(store: NoteStore, userId: string): AsyncGenerator<string> {
  yield HEADER + '\n'
  let after: string | undefined
  do {
    const page = await store.listNotesPage(userId, { after, limit: EXPORT_PAGE_SIZE })
    if (page.notes.length) yield page.notes.map(toCsvRow).join('\n') + '\n'
    after = page.nextCursor
  } while (after)
}
```
