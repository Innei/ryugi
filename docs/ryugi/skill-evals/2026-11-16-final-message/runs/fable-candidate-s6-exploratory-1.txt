    expect(exportNotesCsv([])).toBe(HEADER)
  })

**11. Write** `src/export.ts`

```ts
export interface Note {
  id: string
  title: string
  body: string
  createdAt: string
}

const COLUMNS = ['id', 'title', 'body', 'createdAt'] as const

function escapeField(value: string): string {
  return /[",\r\n]/.test(value) ? `"${value.replace(/"/g, '""')}"` : value
}

export function exportNotesCsv(notes: Note[]): string {
  const rows = notes.map((note) => COLUMNS.map((c) => escapeField(note[c])).join(','))
  return [COLUMNS.join(','), ...rows].join('\r\n')
}
```

Assumption: the spec defines `Note` with these four string fields and the repo has no existing `Note` type to import.
