   ```ts
         title: 'First',
         id: '1',

       expect(exportNotesCsv(notes)).toBe(
         'id,title,body\r\n1,First,你好\r\n',
       );
       expect(notes).toEqual([{ body: '你好', title: 'First', id: '1' }]);
     });

     it.each([
       ['plain', 'plain'],
       ['a,b', '"a,b"'],
       ['a"b', '"a""b"'],
       ['a\nb', '"a\nb"'],
       ['a\rb', '"a\rb"'],
       ['a\r\nb', '"a\r\nb"'],
       ['a,"b"\r\nc', '"a,""b""\r\nc"'],
       ['', ''],
     ])('encodes %j in every column', (value, encoded) => {
       expect(
         exportNotesCsv([{ id: value, title: value, body: value }]),
       ).toBe(`id,title,body\r\n${encoded},${encoded},${encoded}\r\n`);
   ```

    ```ts
    export interface ExportNote {
      readonly id: string;
      readonly title: string;
      readonly body: string;
    }

    function escapeField(value: string): string {
      return /[",\r\n]/.test(value)
        ? `"${value.replace(/"/g, '""')}"`
        : value;
    }

    export function exportNotesCsv(notes: readonly ExportNote[]): string {
      const rows = [
        'id,title,body',
        ...notes.map((note) =>
          [note.id, note.title, note.body].map(escapeField).join(','),
        ),
      ];

      return `${rows.join('\r\n')}\r\n`;
    }
    ```
