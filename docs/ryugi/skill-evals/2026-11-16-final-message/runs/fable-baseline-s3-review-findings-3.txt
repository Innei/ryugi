```ts
export const EXPORT_PAGE_SIZE = 500

export async function* exportNotes(store: NoteStore, userId: string): AsyncGenerator<string> {
  yield HEADER + '\n'
  let after: string | undefined
  for (;;) {
    const page = await store.listNotes(userId, { after, limit: EXPORT_PAGE_SIZE })
    if (page.length === 0) return
    yield page.map(toCsvRow).join('\n') + '\n'
    if (page.length < EXPORT_PAGE_SIZE) return
    after = page[page.length - 1].id
  }
}
```
