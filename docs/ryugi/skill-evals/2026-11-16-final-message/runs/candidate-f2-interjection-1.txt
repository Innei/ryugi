```tsx
import type { Slide } from '../model/slide';
import { SlidePreview } from './SlidePreview';

type PresenterWindowProps = {
  slides: readonly Slide[];
  currentIndex: number;
};

export function PresenterWindow({
  slides,
  currentIndex,
}: PresenterWindowProps) {
  const currentSlide = slides[currentIndex];
  const nextSlide = slides[currentIndex + 1];

  return (
    <main
      aria-label="Presenter view"
      style={{
        display: 'grid',
        gridTemplateColumns: 'minmax(0, 2fr) minmax(0, 1fr)',
        gap: '1rem',
        padding: '1rem',
      }}
    >
      <section aria-label="Current slide" style={{ minWidth: 0 }}>
        <h1>Current slide</h1>
        {currentSlide ? (
          <SlidePreview slide={currentSlide} />
        ) : (
          <p>No current slide</p>
        )}
      </section>

      <section aria-label="Next slide" style={{ minWidth: 0 }}>
        <h2>Next slide</h2>
        {nextSlide ? (
          <SlidePreview slide={nextSlide} />
        ) : (
          <p>End of deck</p>
        )}
      </section>

      <section
        aria-label="Speaker notes"
        style={{ gridColumn: '1 / -1', minWidth: 0 }}
      >
        <h2>Speaker notes</h2>
        <p style={{ whiteSpace: 'pre-wrap', overflowWrap: 'anywhere' }}>
          {currentSlide?.notes || 'No speaker notes'}
        </p>
      </section>
    </main>
  );
}
```
