**Call 12 — write `src/presenter/PresenterWindow.tsx`:**

```tsx
import type { Deck } from '../deck/types'
import { SlideView } from '../viewer/SlideView'

export interface PresenterWindowProps {
  deck: Deck
  currentIndex: number
  elapsedMs: number
}

function formatElapsed(ms: number): string {
  const total = Math.max(0, Math.floor(ms / 1000))
  const mm = String(Math.floor(total / 60)).padStart(2, '0')
  const ss = String(total % 60).padStart(2, '0')
  return `${mm}:${ss}`
}

export function PresenterWindow({ deck, currentIndex, elapsedMs }: PresenterWindowProps) {
  const current = deck.slides[currentIndex]
  const next = deck.slides[currentIndex + 1]

  return (
    <div
      style={{
        display: 'grid',
        gridTemplateColumns: '2fr 1fr',
        gridTemplateRows: '1fr 1fr auto',
        gridTemplateAreas: '"current next" "current notes" "bar bar"',
        gap: 12,
        height: '100vh',
        padding: 12,
      }}
    >
      <section data-testid="presenter-current" style={{ gridArea: 'current' }}>
        <SlideView slide={current} theme={deck.theme} />
      </section>
      <section data-testid="presenter-next" style={{ gridArea: 'next' }}>
        {next ? <SlideView slide={next} theme={deck.theme} /> : <p>End of deck</p>}
      </section>
      <section data-testid="presenter-notes" style={{ gridArea: 'notes', overflowY: 'auto' }}>
        {current.notes}
      </section>
      <footer style={{ gridArea: 'bar', display: 'flex', justifyContent: 'space-between' }}>
        <span data-testid="presenter-timer">{formatElapsed(elapsedMs)}</span>
        <span data-testid="presenter-counter">
          {currentIndex + 1} / {deck.slides.length}
        </span>
      </footer>
    </div>
  )
}
```
