**Call 8 (edit `src/export.ts`)**: replace the body at line 40.
```ts
export const EXPORT_PAGE_SIZE = 1000

export async function* exportNotes(userId: string, repo: NoteRepo): AsyncGenerator<string> {
  yield HEADER + '\n'
  let after: PageArgs['after'] = null
  for (;;) {
    const page = await repo.listByUserPage(userId, { after, limit: EXPORT_PAGE_SIZE })
    if (page.length === 0) return
    yield page.map(toCsvRow).join('\n') + '\n'
    if (page.length < EXPORT_PAGE_SIZE) return
    const last = page[page.length - 1]
    after = { createdAt: last.createdAt, id: last.id }
  }
}
```
