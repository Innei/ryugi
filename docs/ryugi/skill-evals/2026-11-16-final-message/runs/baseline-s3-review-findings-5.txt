import { createHash } from "node:crypto";
import { Writable } from "node:stream";
import { expect, test } from "vitest";
import { exportNotes } from "./export";
import type { NoteRepository } from "./notes";

test("exports 200000 notes with bounded buffering and backpressure", async () => {
  const count = 200_000;
  const body = "x".repeat(1024);
  const updatedAt = "2026-11-10T00:00:00.000Z";
  const expected = createHash("sha256");
  const actual = createHash("sha256");
  let produced = 0;
  let written = 0;
  let rows = 0;

  expected.update("id,title,body,updatedAt\n");
  for (let id = 0; id < count; id++) {
    expected.update(`${id},Note ${id},${body},${updatedAt}\n`);
  }

  async function* iterateNotes() {
    for (let id = 0; id < count; id++) {
      produced++;
      if (produced - written > 128) {
        throw new Error("more than 128 notes buffered");
      }
      yield {
        id: String(id),
        title: `Note ${id}`,
        body,
        updatedAt,
      };
    }
  }

  const repository: NoteRepository = {
    iterateNotes,
    async listNotes() {
      const notes = [];
      for await (const note of iterateNotes()) notes.push(note);
      return notes;
    },
  };

  const destination = new Writable({
    highWaterMark: 1024,
    write(chunk, _encoding, callback) {
      const text = chunk.toString();
      actual.update(chunk);
      rows += text.split("\n").length - 1;
      written = Math.max(0, rows - 1);
      setImmediate(callback);
    },
  });

  await exportNotes(repository, destination);

  expect(produced).toBe(count);
  expect(written).toBe(count);
  expect(actual.digest("hex")).toBe(expected.digest("hex"));
});

test("exports an empty account as a header-only CSV", async () => {
  const repository: NoteRepository = {
    async *iterateNotes() {},
    async listNotes() {
      return [];
    },
  };
  let csv = "";

  await exportNotes(
    repository,
    new Writable({
      write(chunk, _encoding, callback) {
        csv += chunk.toString();
        callback();
      },
    }),
  );

  expect(csv).toBe("id,title,body,updatedAt\n");
});
```

export async function exportNotes(
  repository: NoteRepository,
  destination: Writable,
): Promise<void> {
  async function* rows() {
    yield "id,title,body,updatedAt\n";

    for await (const note of repository.iterateNotes()) {
      yield (
        [note.id, note.title, note.body, note.updatedAt]
          .map(csvCell)
          .join(",") + "\n"
      );
    }
  }

  await pipeline(rows(), destination);
}
```
