```ts
describe('exportNotes on a large account', () => {
  const TOTAL = 5000

  function pagedStore() {
    const calls = { all: 0, limits: [] as number[] }
    const store = {
      listAllNotes: async () => {
        calls.all++
        return Array.from({ length: TOTAL }, (_, i) => makeNote(i))
      },
      listNotes: async (_userId: string, { cursor, limit }: { cursor?: string; limit: number; order: 'asc' }) => {
        calls.limits.push(limit)
        const start = cursor ? Number(cursor) : 0
        const end = Math.min(start + limit, TOTAL)
        const notes = Array.from({ length: end - start }, (_, i) => makeNote(start + i))
        return { notes, nextCursor: end < TOTAL ? String(end) : null }
      },
    }
    return { store, calls }
  }

  it('reads notes in bounded pages and never loads the whole account', async () => {
    const { store, calls } = pagedStore()
    const chunks: string[] = []
    const out = new Writable({ write(c, _e, cb) { chunks.push(String(c)); cb() } })
    await exportNotes(store as any, 'u1', out)
    expect(calls.all).toBe(0)
    expect(calls.limits.length).toBeGreaterThan(1)
    expect(Math.max(...calls.limits)).toBeLessThanOrEqual(500)
    expect(chunks.join('').trimEnd().split('\n')).toHaveLength(TOTAL + 1) // header + rows
  })

  it('waits for a slow reader instead of buffering the export', async () => {
    const { store } = pagedStore()
    let peak = 0
    const out = new Writable({
      highWaterMark: 16 * 1024,
      write(_c, _e, cb) { peak = Math.max(peak, out.writableLength); setImmediate(cb) },
    })
    await exportNotes(store as any, 'u1', out)
    expect(peak).toBeLessThan(256 * 1024)
  })
})
```
