```text
+type ExportNote = {
+  title: string;
+  content: string;
+  createdAt: string;
+};
+
+function escapeCsvField(value: string): string {
+  return /[",\r\n]/.test(value)
+    ? `"${value.replace(/"/g, '""')}"`
+    : value;
+}
+
+export function exportNotesCsv(notes: readonly ExportNote[]): string {
+  return [
+    'title,content,createdAt',
+    ...notes.map((note) =>
+      [note.title, note.content, note.createdAt]
+        .map(escapeCsvField)
+        .join(','),
+    ),
+  ].join('\r\n');
+}
*** End Patch
```
