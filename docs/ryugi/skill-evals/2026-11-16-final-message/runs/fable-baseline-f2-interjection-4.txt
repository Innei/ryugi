**Call 12 — write** `src/presenter/PresenterWindow.tsx`
```tsx
import { SlideSurface } from '../viewer/SlideSurface'
import { useDeckStore } from '../store/deckStore'

export function PresenterWindow() {
  const slides = useDeckStore((s) => s.slides)
  const currentIndex = useDeckStore((s) => s.currentIndex)
  const current = slides[currentIndex]
  const next = slides[currentIndex + 1]

  return (
    <div
      style={{
        display: 'grid',
        gridTemplateColumns: '2fr 1fr',
        gridTemplateRows: 'auto 1fr auto',
        gridTemplateAreas: '"current next" "current notes" "footer footer"',
        gap: 16,
        height: '100vh',
        padding: 16,
      }}
    >
      <section data-testid="presenter-current" style={{ gridArea: 'current' }}>
        {current && <SlideSurface slide={current} />}
      </section>
      <section data-testid="presenter-next" style={{ gridArea: 'next' }}>
        {next ? <SlideSurface slide={next} /> : 'End of deck'}
      </section>
      <section data-testid="presenter-notes" style={{ gridArea: 'notes', overflowY: 'auto' }}>
        {current?.notes ?? ''}
      </section>
      <footer style={{ gridArea: 'footer', display: 'flex', justifyContent: 'space-between' }}>
        <span data-testid="presenter-counter">{`${currentIndex + 1} / ${slides.length}`}</span>
        <span data-testid="presenter-timer-slot" />
      </footer>
    </div>
  )
}
```
