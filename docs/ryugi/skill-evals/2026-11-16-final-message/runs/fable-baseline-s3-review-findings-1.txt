```ts
async function collect(chunks: AsyncIterable<string>): Promise<string> {
  let out = ''
  for await (const c of chunks) out += c
  return out
}

function pagedStore(total: number) {
  const notes = Array.from({ length: total }, (_, i) => makeNote({ id: String(i).padStart(6, '0') }))
  const limits: number[] = []
  return {
    limits,
    listNotes: async () => { throw new Error('unbounded listNotes called') },
    listNotesPage: async (_u: string, { after, limit }: { after?: string; limit: number }) => {
      limits.push(limit)
      const start = after ? notes.findIndex(n => n.id === after) + 1 : 0
      return notes.slice(start, start + limit)
    },
  }
}

it('exportNotes reads notes in bounded pages, never the whole account', async () => {
  const store = pagedStore(2500)
  const it = exportNotes('u1', store as any)[Symbol.asyncIterator]()
  await it.next(); await it.next()            // header + first page only
  expect(store.limits).toHaveLength(1)        // lazy: nothing fetched ahead of the consumer
  const csv = await collect(exportNotes('u1', store as any))
  expect(csv.trimEnd().split('\n')).toHaveLength(2501)   // header + 2500 rows
  expect(Math.max(...store.limits)).toBeLessThanOrEqual(500)
})
```

```ts
const PAGE_SIZE = 500

export async function* exportNotes(userId: string, store: NoteStore): AsyncGenerator<string> {
  yield HEADER + '\n'
  let after: string | undefined
  for (;;) {
    const page = await store.listNotesPage(userId, { after, limit: PAGE_SIZE })
    if (page.length === 0) return
    yield page.map(toCsvRow).join('\n') + '\n'
    if (page.length < PAGE_SIZE) return
    after = page[page.length - 1].id
  }
}
```
