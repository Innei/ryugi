```tsx
import { SlideView } from '../viewer/SlideView'
import type { Deck } from '../deck/types'

export function PresenterLayout({ deck, currentIndex }: { deck: Deck; currentIndex: number }) {
  const current = deck.slides[currentIndex]
  const next = deck.slides[currentIndex + 1]
  return (
    <div style={{ display: 'grid', gridTemplateColumns: '2fr 1fr', gridTemplateRows: '1fr 1fr', gap: 16, height: '100%' }}>
      <section data-testid="current-slide" style={{ gridRow: '1 / span 2' }}>
        <SlideView slide={current} theme={deck.theme} />
      </section>
      <section data-testid="next-slide">
        {next ? <SlideView slide={next} theme={deck.theme} /> : <p>End of deck</p>}
      </section>
      <section data-testid="speaker-notes" style={{ overflowY: 'auto' }}>
        {current.notes}
      </section>
    </div>
  )
}
```
