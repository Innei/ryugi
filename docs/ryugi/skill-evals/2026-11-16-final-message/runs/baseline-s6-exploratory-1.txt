10. **Shell call — implement**

    ```sh
    cat > src/export.ts <<'EOF'
    export function exportNotesCsv(
      notes: readonly { title: string; body: string }[],
    ): string {
      const escapeField = (value: string): string =>
        /[",\r\n]/.test(value)
          ? `"${value.replace(/"/g, '""')}"`
          : value;

      return [
        "title,body",
        ...notes.map(
          ({ title, body }) => `${escapeField(title)},${escapeField(body)}`,
        ),
      ].join("\r\n");
    }
    EOF
    ```

    Assumed result: exit 0, no output.
