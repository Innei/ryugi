   ```ts
   import { describe, expect, it } from 'vitest';
   import { exportNotes, type ExportNote } from '../src/export';

   function note(id: number): ExportNote {
     return {
       id: String(id),
       title: 'Title',
       body: 'Body',
       updatedAt: '2026-11-10T00:00:00Z',
     };
   }

   describe('streaming CSV export', () => {
     it('writes notes before consuming the entire source', async () => {
       let received = '';
       const destination = new Writable({
         write(chunk, _encoding, callback) {
           received += chunk.toString();
           callback();
         },
       });

       async function* notes() {
         yield note(0);
         for (let id = 1; id < 1_024; id++) {
           yield note(id);
         }
         expect(received).toContain(
           '0,Title,Body,2026-11-10T00:00:00Z\r\n',
         );
         yield note(1_024);
       }

       await exportNotes(notes(), destination);

       expect(received.startsWith('id,title,body,updatedAt\r\n')).toBe(true);
       expect(received).toContain(
         '1024,Title,Body,2026-11-10T00:00:00Z\r\n',
       );
     });

     it('keeps source consumption bounded behind a slow destination', async () => {
       let produced = 0;
       let written = 0;
       let maximumLead = 0;
       let headerWritten = false;

       const destination = new Writable({
         highWaterMark: 1,
         write(_chunk, _encoding, callback) {
           if (headerWritten) {
             written++;
           } else {
             headerWritten = true;
           }
           setImmediate(callback);
         },
       });

       async function* notes() {
         for (let id = 0; id < 200_000; id++) {
           produced++;
           maximumLead = Math.max(maximumLead, produced - written);
           yield note(id);
         }
       }

       await exportNotes(notes(), destination);

       expect(written).toBe(200_000);
       expect(maximumLead).toBeLessThan(8_192);
     });
   });
   ```
