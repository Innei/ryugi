+import type { ReactNode } from 'react'

+type PresenterLayoutProps = {
+  currentSlide: ReactNode
+  nextSlide: ReactNode
+  notes: string

+
+export function PresenterLayout({
+  currentSlide,
+  nextSlide,
+  notes,
+}: PresenterLayoutProps) {
+  return (
+    <main
+      style={{
+        display: 'grid',
+        gridTemplateColumns: 'minmax(0, 2fr) minmax(0, 1fr)',
+        gridTemplateRows: 'minmax(0, 1fr) minmax(0, 1fr)',
+        gap: 16,
+        height: '100%',
+        minHeight: 0,
+        padding: 16,
+        boxSizing: 'border-box',
+      }}
+    >
+      <section
+        aria-label="当前页"
+        style={{
+          gridRow: '1 / 3',
+          minWidth: 0,
+          minHeight: 0,
+          overflow: 'auto',
+        }}
+      >
+        {currentSlide}
+      </section>
+      <section
+        aria-label="下一页"
+        style={{ minWidth: 0, minHeight: 0, overflow: 'auto' }}
+      >
+        {nextSlide ?? <p>已是最后一页</p>}
+      </section>
+      <section
+        aria-label="演讲备注"
+        style={{
+          minWidth: 0,
+          minHeight: 0,
+          overflow: 'auto',
+          whiteSpace: 'pre-wrap',
+          overflowWrap: 'anywhere',
+        }}
+      >
+        {notes || '暂无备注'}
+      </section>
+    </main>
+  )
+}
