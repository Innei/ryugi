    ```sh
    type Note = { title: string; body: string }

    function escapeCsvField(value: string): string {
      if (/[",\r\n]/.test(value)) {
        return `"${value.replace(/"/g, '""')}"`
      }
      return value
    }

    export function exportNotesCsv(notes: readonly Note[]): string {
      const rows = notes.map(({ title, body }) =>
        `${escapeCsvField(title)},${escapeCsvField(body)}\r\n`,
      )
      return `title,body\r\n${rows.join('')}`
    }
    EOF
    ```
