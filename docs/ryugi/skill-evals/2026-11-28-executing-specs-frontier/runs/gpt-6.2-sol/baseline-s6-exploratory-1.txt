   export interface ExportNote {
     id: string
     title: string
     body: string
   }

   function escapeCsvField(value: string): string {
     return /[",\r\n]/.test(value) ? `"${value.replace(/"/g, '""')}"` : value
   }

   export function exportNotesCsv(notes: readonly ExportNote[]): string {
     const rows = notes.map(({ id, title, body }) =>
       [id, title, body].map(escapeCsvField).join(','),
     )

     return ['id,title,body', ...rows, ''].join('\r\n')
   }
   EOF
