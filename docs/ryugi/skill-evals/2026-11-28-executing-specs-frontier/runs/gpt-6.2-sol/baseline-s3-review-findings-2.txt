### One fix pass

6. **Shell call**  
   Replace the eager read and CSV assembly. The existing CSV formatter and response headers are reused.

   ```sh
   python3 - <<'PY'
   from pathlib import Path

   path = Path("src/export.ts")
   source = path.read_text()
   before = '''  const notes = await repo.listNotes(accountId);
     const csv = CSV_HEADER + notes.map(formatCsvRow).join("");
     return new Response(csv, { headers: CSV_HEADERS });
   '''
   after = '''  const pageSize = 500;
     const encoder = new TextEncoder();

     async function* rows() {
       yield CSV_HEADER;
       let after: string | null = null;
       do {
         const page = await repo.listNotesPage(accountId, { after, limit: pageSize });
         for (const note of page.notes) {
           yield formatCsvRow(note);
         }
         after = page.nextCursor;
       } while (after !== null);
     }

     const csvRows = rows();
     const body = new ReadableStream<Uint8Array>({
       async pull(controller) {
         const row = await csvRows.next();
         if (row.done) {
           controller.close();
           return;
         }
         controller.enqueue(encoder.encode(row.value));
       },
       async cancel() {
         await csvRows.return(undefined);
       },
     }, { highWaterMark: 0 });

     return new Response(body, { headers: CSV_HEADERS });
   '''
   assert source.count(before) == 1
   path.write_text(source.replace(before, after, 1))
   PY
   ```

   **Assumed result.** Exit `0`. Existing test fixtures already provide default paginated results from the fixture array, so the other tests can exercise the new read path.
