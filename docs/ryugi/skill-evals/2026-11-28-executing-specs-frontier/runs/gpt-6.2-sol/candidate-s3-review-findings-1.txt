### 2. Reproduce the bulk read

```sh
cat > src/export.review.test.ts <<'EOF'
import { createReadStream } from 'node:fs'
import { mkdtemp, rm } from 'node:fs/promises'
import { tmpdir } from 'node:os'
import { join } from 'node:path'
import { createInterface } from 'node:readline'
import { expect, test } from 'vitest'
import { exportNotes } from './export'

test('exports 200,000 notes in order through bounded repository reads', async () => {
  const directory = await mkdtemp(join(tmpdir(), 'notes-export-test-'))
  const destination = join(directory, 'notes.csv')
  const noteCount = 200_000
  const updatedAt = '2026-11-10T00:00:00.000Z'

  const repo = {
    async listByAccount(): Promise<never> {
      throw new Error('Export attempted to load all notes at once')
    },
    async *iterateByAccount(accountId: string, { batchSize }: { batchSize: number }) {
      expect(accountId).toBe('account-1')
      expect(Number.isInteger(batchSize) && batchSize > 0 && batchSize <= 1_000).toBe(true)
      for (let start = 0; start < noteCount; start += batchSize) {
        const page = Array.from(
          { length: Math.min(batchSize, noteCount - start) },
          (_, offset) => {
            const id = String(start + offset)
            return { id, title: `Note ${id}`, body: `Body ${id}`, updatedAt }
          },
        )
        yield* page
      }
    },
  }

  try {
    await exportNotes(repo, 'account-1', destination)
    const lines = createInterface({
      input: createReadStream(destination),
      crlfDelay: Infinity,
    })
    let lineNumber = 0
    for await (const line of lines) {
      const id = lineNumber - 1
      const expected = lineNumber === 0
        ? 'id,title,body,updatedAt'
        : `${id},Note ${id},Body ${id},${updatedAt}`
      if (line !== expected) expect(line).toBe(expected)
      lineNumber++
    }
    expect(lineNumber).toBe(noteCount + 1)
  } finally {
    await rm(directory, { recursive: true, force: true })
  }
}, 30_000)
EOF
```
