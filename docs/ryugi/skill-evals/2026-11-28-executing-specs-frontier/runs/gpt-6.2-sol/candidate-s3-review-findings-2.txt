```ts
it('streams notes one page at a time in export order', async () => {
  const notes = makeNotes(2_001)
  const repository = makeNotesRepository(notes)
  const listAll = vi.spyOn(repository, 'listAll')
  const listPage = vi.spyOn(repository, 'listPage')
  const rows = exportNotes(repository, accountId)

  try {
    const header = await rows.next()
    expect(header.done).toBe(false)
    expect(await rows.next()).toEqual({
      done: false,
      value: encodeNote(notes[0]),
    })
    expect(listAll).toHaveBeenCalledTimes(0)
    expect(listPage).toHaveBeenCalledTimes(1)
    expect(listPage).toHaveBeenCalledWith(accountId, {
      after: null,
      limit: 1_000,
    })

    const remainingRows = []
    for await (const row of rows) {
      remainingRows.push(row)
    }
    expect(remainingRows).toEqual(notes.slice(1).map(encodeNote))
    expect(listPage).toHaveBeenCalledTimes(3)
  } finally {
    await rows.return(undefined)
  }
})
```
