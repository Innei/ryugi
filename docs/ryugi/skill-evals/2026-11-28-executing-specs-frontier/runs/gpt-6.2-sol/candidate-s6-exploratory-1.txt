11. **Shell:**
    ```sh
    cat > src/export.ts <<'EOF'
    interface ExportNote {
      title: string
      body: string
    }

    function escapeCsvField(value: string): string {
      if (/[",\r\n]/.test(value)) {
        return `"${value.replace(/"/g, '""')}"`
      }

      return value
    }

    export function exportNotesCsv(notes: readonly ExportNote[]): string {
      const rows = notes.map(({ title, body }) =>
        [escapeCsvField(title), escapeCsvField(body)].join(','),
      )

      return ['title,body', ...rows, ''].join('\n')
    }
    EOF
    pnpm exec vitest run src/export.test.ts
    ```

    Assumed the full output shows exit 0 and three passing tests.
