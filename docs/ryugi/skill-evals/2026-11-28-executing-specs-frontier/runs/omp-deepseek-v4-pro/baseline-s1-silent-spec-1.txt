### 4. Implement

```ts
import type { Note } from './types';

function csvField(value: string): string {
  return /[",\r\n]/.test(value)
    ? `"${value.replace(/"/g, '""')}"`
    : value;
}

export function toCsvRow(note: Note): string {
  return [note.id, note.title, note.updatedAt].map(csvField).join(',');
}
```
