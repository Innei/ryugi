export const CSV_HEADER = "id,title,body,createdAt,updatedAt";

export function noteToCsvRow(note: Note): string {
  return [note.id, note.title, note.body, note.createdAt, note.updatedAt]

function escapeCsv(v: string): string {

Fix — `src/export.ts` (drop `getAllNotes`, stream through a sink):

```ts
export interface NoteSink {
  write(line: string): void;
}

export async function exportNotes(
  notes: AsyncIterable<Note>,
  sink: NoteSink,
): Promise<number> {
  sink.write(CSV_HEADER);
  let count = 0;
  for await (const note of notes) {
    sink.write(noteToCsvRow(note));
    count += 1;
  }
  return count;
}
```

Add the streaming primitive to `src/notes.ts`:

```ts
export const NOTES_PAGE_SIZE = 500;

export async function* iterateNotes(): AsyncGenerator<Note> {
  let cursor: string | undefined;
  do {
    const page = await listNotesPage(NOTES_PAGE_SIZE, cursor);
    for (const note of page.notes) yield note;
    cursor = page.nextCursor ?? undefined;
  } while (cursor !== undefined);
}
```

Migrate the caller `src/cli/export.ts` to stream to the file:
