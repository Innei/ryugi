### 2.2 GREEN: page the reads

`src/notes/db.ts` — replace `getAllNotes` with a keyset-paged reader (only caller was export):

```ts
export interface NotePage {
  items: Note[];
  nextCursor: string | null;
}

export const EXPORT_PAGE_SIZE = 500;

export async function listNotes(opts: {
  cursor: string | null;
  limit: number;
}): Promise<NotePage> {
  const { cursor, limit } = opts;
  const rows = await query(
    'SELECT id, title, body, created_at, updated_at FROM notes' +
      (cursor ? ' WHERE id > ?' : '') +
      ' ORDER BY id ASC LIMIT ?',
    ...(cursor ? [cursor] : []),
    limit + 1,
  );
  const items = rows.slice(0, limit).map(rowToNote);
  return {
    items,
    nextCursor: rows.length > limit ? items[items.length - 1].id : null,
  };
}
```

`src/export.ts` — stream page by page; at most one page of `Note` objects lives at a time:

```ts
import { listNotes, EXPORT_PAGE_SIZE, type Note } from '../notes/db';

export async function exportNotes(): Promise<string> {
  const rows: string[] = [CSV_HEADER.join(',')];
  let cursor: string | null = null;
  do {
    const page = await listNotes({ cursor, limit: EXPORT_PAGE_SIZE });
    for (const note of page.items) rows.push(noteToRow(note));
    cursor = page.nextCursor;
  } while (cursor !== null);
  return rows.join('\n');
}
```
