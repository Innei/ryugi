**Implement** the RFC 4180 row writer:

```ts
// src/export.ts
export interface Note { id: string; title: string; updatedAt: string; }

function csvField(value: string): string {
  return /[",\r\n]/.test(value) ? `"${value.replace(/"/g, '""')}"` : value;
}

export function notesToCsv(notes: Note[]): string {
  if (notes.length === 0) return '';
  return notes
    .map((n) => [n.id, n.title, n.updatedAt].map(csvField).join(','))
    .join('\n') + '\n';
}
```
