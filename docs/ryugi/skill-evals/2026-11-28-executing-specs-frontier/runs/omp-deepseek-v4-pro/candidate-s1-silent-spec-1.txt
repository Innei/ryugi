**Implement** `src/export/csv.ts`:

```ts
export type ExportNote = {
  id: string;
  title: string;
  updatedAt: string;
};

const needsQuoting = /[",\r\n]/;

function quoteField(value: string): string {
  if (needsQuoting.test(value)) {
    return `"${value.replace(/"/g, '""')}"`;
  }
  return value;
}

export function toRow(note: ExportNote): string {
  return [note.id, note.title, note.updatedAt].map(quoteField).join(',');
}
```
