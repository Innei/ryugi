`src/export.ts` (new):

```ts
import { Note, iterateNotes } from './notes';

export const CSV_HEADER = 'id,title,updatedAt';

function escapeField(value: string): string {
  if (/[",\n]/.test(value)) return '"' + value.replace(/"/g, '""') + '"';
  return value;
}

export function toCsvRow(note: Note): string {
  return [note.id, note.title, note.updatedAt].map(escapeField).join(',');
}

export interface NoteSink { write(chunk: string): void; }

export async function exportNotes(
  sink: NoteSink,
  notes: AsyncIterable<Note> = iterateNotes(),
): Promise<void> {
  sink.write(CSV_HEADER + '\n');
  for await (const note of notes) {
    sink.write(toCsvRow(note) + '\n');
  }
}
```
