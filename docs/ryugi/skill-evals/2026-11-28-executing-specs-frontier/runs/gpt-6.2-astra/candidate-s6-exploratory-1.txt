**9. Implement the export.**

```sh
cat > src/export.ts <<'EOF'
function csvField(value: string): string {
  if (/[",\r\n]/.test(value)) {
    return `"${value.replace(/"/g, '""')}"`
  }
  return value
}

export function exportNotesCsv(
  notes: ReadonlyArray<{ title: string; body: string }>,
): string {
  const records = notes.map(({ title, body }) =>
    [csvField(title), csvField(body)].join(','),
  )
  return ['title,body', ...records].join('\r\n')
}
EOF
```
