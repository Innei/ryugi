**6. Implement the serializer and run Verify.** Simulated shell call.

```sh
cat > src/export.ts <<'EOF'
function csvField(value: string): string {
  if (/[",\r\n]/.test(value)) {
    return `"${value.replace(/"/g, '""')}"`
  }
  return value
}

export function exportNotesCsv(
  notes: readonly { title: string; body: string }[],
): string {
  const rows = notes.map(({ title, body }) => `${csvField(title)},${csvField(body)}`)
  return ['title,body', ...rows].join('\n')
}
EOF
pnpm test
```
