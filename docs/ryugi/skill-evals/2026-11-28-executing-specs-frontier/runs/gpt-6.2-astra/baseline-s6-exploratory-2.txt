**5. Implement the format and run Verify.**

```sh
cat > src/export.ts <<'EOF'
function csvField(value: string): string {
  if (/[",\r\n]/.test(value)) {
    return `"${value.replace(/"/g, '""')}"`
  }
  return value
}

export function exportNotesCsv(notes: readonly { title: string; body: string }[]): string {
  const rows = notes.map(note => `${csvField(note.title)},${csvField(note.body)}`)
  return ['title,body', ...rows].join('\r\n')
}
EOF
pnpm test
```
