    from pathlib import Path

    def now():
        return datetime.datetime.now(datetime.timezone.utc).isoformat(timespec="seconds").replace("+00:00", "Z")

    def curl(args, config=None):
        result = subprocess.run(
            ["curl", "--silent", "--show-error", "--max-time", "15", *args],
            input=config, capture_output=True, text=True)
        if result.stderr:
            print(result.stderr, flush=True)
        result.check_returncode()
        print(result.stdout, flush=True)
        body, status = result.stdout.rsplit("\n", 1)
        return body, int(status)

    key = os.environ["PROD_TEST_KEY"]
    assert key and "\n" not in key and "\r" not in key
    escaped_key = key.replace("\\", "\\\\").replace('"', '\\"')
    config = 'header = "Authorization: Bearer ' + escaped_key + '"\n'
    fixture = Path("fixtures/messages-smoke.json")
    json.loads(fixture.read_text())

    print("capturedAt=" + now(), flush=True)
    print("Environment: https://api.example.com; build 7c1d9e4", flush=True)
    print("Expected before measuring: requests 1–60 = HTTP 200; request 61 = HTTP 429.", flush=True)
    print("Dedicated test key idle for at least 60 seconds, per project instructions.", flush=True)
    print("Request template: curl --config - --silent --show-error --max-time 15 "
          "--request POST --header 'Content-Type: application/json' "
          "--data-binary @fixtures/messages-smoke.json "
          "--write-out '\\n%{http_code}' https://api.example.com/v1/messages", flush=True)
    print("Authorization configuration supplied on stdin; secret omitted.", flush=True)

    started = time.monotonic()
    for number in range(1, 62):
        print(f"request={number} capturedAt={now()}", flush=True)
        body, status = curl([
            "--config", "-", "--request", "POST",
            "--header", "Content-Type: application/json",
            "--data-binary", "@" + str(fixture),
            "--write-out", "\n%{http_code}",
            "https://api.example.com/v1/messages"
        ], config)
        expected = 200 if number <= 60 else 429
        assert status == expected, (number, status, expected)

    elapsed = time.monotonic() - started
    print(f"elapsed_seconds={elapsed:.3f}", flush=True)
    assert elapsed < 60, "Burst crossed the rolling window"
    print("PASS: requests 1–60 returned 200; request 61 returned 429 within 60 seconds.", flush=True)

    print("Neighbour: POST /v1/messages without Authorization; expected 401.", flush=True)
