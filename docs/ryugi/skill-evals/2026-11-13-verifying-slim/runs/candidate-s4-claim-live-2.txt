    ```sh
    set -o pipefail
    python3 - <<'PY' 2>&1 | tee .ryugi/artifacts/verify/rate-limit/evidence/production.txt
    import datetime
    import json
    import os
    import pathlib
    import subprocess
    import time

    def stamp():
        return datetime.datetime.now(datetime.timezone.utc).isoformat(timespec="seconds").replace("+00:00", "Z")

    expected = subprocess.check_output(
        ["git", "rev-parse", "7c1d9e4"], text=True
    ).strip()
    key = os.environ["PROD_TEST_KEY"]
    assert key and "\n" not in key and "\r" not in key
    fixture = pathlib.Path("fixtures/messages-valid.json").read_text()
    json.loads(fixture)

    def health():
        raw = subprocess.check_output([
            "curl", "--fail-with-body", "--silent", "--show-error",
            "--max-time", "10", "https://api.example.com/health"
        ], text=True)
        print(stamp(), "GET /health", raw, flush=True)
        assert json.loads(raw)["build"] == expected

    def message(authenticated):
        args = [
            "curl", "--silent", "--show-error", "--max-time", "10",
            "--request", "POST", "--header", "Content-Type: application/json",
            "--data-binary", "@fixtures/messages-valid.json",
            "--output", "/dev/null", "--write-out", "%{http_code}",
        ]
        config = ""
        if authenticated:
            escaped = key.replace("\\", "\\\\").replace('"', '\\"')
            args += ["--config", "-"]
            config = 'header = "Authorization: Bearer ' + escaped + '"\n'
        args.append("https://api.example.com/v1/messages")
        result = subprocess.run(
            args, input=config, text=True, capture_output=True, check=True
        )
        if result.stderr:
            print(result.stderr, end="", flush=True)
        return int(result.stdout)

    print(stamp(), "production", expected, flush=True)
    print("Expected: authenticated requests 1-60 => 200; request 61 => 429; unauthenticated => 401", flush=True)
    print("Fixture: fixtures/messages-valid.json; dedicated unused test key; no retries", flush=True)
    health()

    wait = 60 - time.time() % 60 + 0.25
    print(stamp(), "Waiting for a fresh UTC minute:", round(wait, 3), "seconds", flush=True)
    time.sleep(wait)
    window = int(time.time() // 60)
    started = time.monotonic()

    for number in range(1, 62):
        assert int(time.time() // 60) == window, "Minute changed before request"
        status = message(True)
        print(stamp(), "POST /v1/messages", "request", number, "status", status, flush=True)
        assert int(time.time() // 60) == window, "Minute changed during request"
        assert status == (200 if number <= 60 else 429), (number, status)

    print("Boundary assertions passed; elapsed seconds:", round(time.monotonic() - started, 3), flush=True)
    status = message(False)
    print(stamp(), "POST /v1/messages unauthenticated status", status, flush=True)
    assert status == 401
    health()
    print(stamp(), "PASS: production build, 60 successes, request 61 rejected, authentication preserved, health preserved", flush=True)
    PY
    ```
