12. **Write file** `/Users/p/git/notes-app/.ryugi/artifacts/verify/csv-export/evidence/check_csv.py`:
    ```python
    import csv
    import json
    from pathlib import Path

    root = Path(__file__).parent
    notes = json.loads((root / "notes.json").read_text())
    with (root / "notes.csv").open(encoding="utf-8-sig", newline="") as f:
        reader = csv.DictReader(f)
        assert reader.fieldnames == ["id", "title", "content"], reader.fieldnames
        rows = list(reader)

    def values(note):
        return tuple(str(note[key]) for key in ("id", "title", "content"))

    assert len(notes) == 42, len(notes)
    assert len(rows) == 42, len(rows)
    assert sorted(map(values, rows)) == sorted(map(values, notes))
    print("PASS: downloaded CSV has id,title,content headers")
    print("PASS: 42 downloaded records exactly match the 42 local notes")
    ```
