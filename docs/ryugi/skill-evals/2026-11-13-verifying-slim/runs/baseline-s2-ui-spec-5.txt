8. **Browser evaluate call** — exercise both directions on each switch, inspect actual animation, check the request and persisted state, and restore the original preference:

   ```js
   async () => {
     const results = [];
     const initial = await fetch('/api/prefs').then(r => r.json());

     for (const [field, label] of [
       ['emailNotifications', 'Email notifications'],
       ['pushNotifications', 'Push notifications']
     ]) {
       for (const expected of [!initial[field], initial[field]]) {
         const control = [...document.querySelectorAll('[role="switch"]')]
           .find(el => el.getAttribute('aria-label') === label);
         const thumb = document.querySelector(`[data-testid="${field}-thumb"]`);
         if (!control || !thumb) throw new Error(`Missing ${field}`);

         const patches = [];
         const originalFetch = window.fetch;
         window.fetch = async (...args) => {
           const response = await originalFetch(...args);
           if (String(args[0]).includes('/api/prefs') &&
               args[1]?.method?.toUpperCase() === 'PATCH') {
             patches.push({
               status: response.status,
               body: args[1].body
             });
           }
           return response;
         };

         try {
           const before = getComputedStyle(thumb).transform;
           control.click();
           await new Promise(requestAnimationFrame);
           const animations = thumb.getAnimations();
           if (!animations.length) throw new Error(`No animation: ${field}`);
           const durations = animations.map(a => a.effect.getTiming().duration);
           if (!durations.every(ms => ms === 160))
             throw new Error(`Wrong duration: ${durations}`);
           await Promise.all(animations.map(a => a.finished));
           const after = getComputedStyle(thumb).transform;
           if (before === after) throw new Error(`Thumb did not move: ${field}`);

           const deadline = performance.now() + 3000;
           let prefs;
           do {
             prefs = await originalFetch('/api/prefs').then(r => r.json());
             if (prefs[field] === expected && patches.length) break;
             await new Promise(r => setTimeout(r, 50));
           } while (performance.now() < deadline);

           if (patches.length !== 1 || patches[0].status !== 200 ||
               prefs[field] !== expected) {
             throw new Error(JSON.stringify({field, expected, patches, prefs}));
           }
           const other = field === 'emailNotifications'
             ? 'pushNotifications' : 'emailNotifications';
           if (prefs[other] !== initial[other])
             throw new Error(`Unrelated preference changed: ${other}`);

           results.push({field, expected, patches, durations, before, after});
         } finally {
           window.fetch = originalFetch;
         }
       }
     }
     return results;
   }
   ```

   **Assumed full output:** four successful transitions; every UI-generated PATCH returns 200; each following GET contains the requested value; the other preference stays unchanged. Every thumb has an active 160ms animation and different initial/final transforms. Both original values are restored.

11. **Shell call** — capture raw API evidence, with assertions and original-state restoration:

    ```sh
    cd /Users/p/git/notes-app
    python3 - <<'PY' > .ryugi/artifacts/verify/notification-settings/evidence/persistence.txt 2>&1
    import datetime, json, subprocess

    base = "http://localhost:5173"
    print("Environment: local dev http://localhost:5173 @ a1b2c3d", flush=True)
    print("CapturedAt:", datetime.datetime.now(datetime.timezone.utc).isoformat(), flush=True)

    def request(method, body=None):
        args = ["curl", "--silent", "--show-error", "--fail-with-body",
                "-X", method, base + "/api/prefs",
                "-w", "\n%{http_code}"]
        if body is not None:
            args += ["-H", "Content-Type: application/json", "--data", json.dumps(body)]
        print("$", subprocess.list2cmdline(args), flush=True)
        output = subprocess.check_output(args, text=True)
        print(output, flush=True)
        payload, status = output.rsplit("\n", 1)
        assert status == "200", status
        return json.loads(payload) if payload else None

    initial = request("GET")
    fields = ["emailNotifications", "pushNotifications"]
    try:
        for field in fields:
            for value in [not initial[field], initial[field]]:
                request("PATCH", {field: value})
                saved = request("GET")
                assert saved[field] == value, (field, value, saved)
                for key in initial:
                    if key != field:
                        assert saved[key] == initial[key], (key, saved)
                print("PASS:", field, "=", value,
                      "; following GET matches; other preferences unchanged", flush=True)
    finally:
        request("PATCH", {field: initial[field] for field in fields})
        assert request("GET") == initial
        print("PASS: original preferences restored", flush=True)
    print("PASS: all persistence assertions", flush=True)
    PY
    ```

    **Assumed output:** exit 0.
