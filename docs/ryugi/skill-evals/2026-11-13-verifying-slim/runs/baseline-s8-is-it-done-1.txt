   import csv
   import json
   from pathlib import Path

   evidence = Path(".ryugi/artifacts/verify/csv-export/evidence")
   notes = json.loads((evidence / "notes.json").read_text())
   columns = ["id", "title", "content", "createdAt", "updatedAt"]

   with (evidence / "download.csv").open(newline="", encoding="utf-8-sig") as f:
       reader = csv.DictReader(f)
       assert reader.fieldnames == columns, reader.fieldnames
       rows = list(reader)

   assert len(rows) == 42, len(rows)
   assert len({r["id"] for r in rows}) == 42

       str(n["id"]): {k: str(n[k]) for k in columns}
       for n in notes
