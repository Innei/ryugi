   ```js
   const fs = require('node:fs/promises');
   const path = require('node:path');
   const { chromium } = require('playwright');
   const root = '/Users/p/git/notes-app';
   const art = path.join(root, '.ryugi/artifacts/artboard/settings-redesign');
   const evidence = path.join(root, '.ryugi/artifacts/verify/notification-settings/evidence');

   const browser = await chromium.launch();
   const context = await browser.newContext({
     viewport: { width: 1280, height: 800 },
     reducedMotion: 'no-preference',
     recordVideo: {
       dir: path.join(evidence, 'recording-scratch'),
       size: { width: 1280, height: 800 }
     }
   });
   const page = await context.newPage();
   const lines = [];
   const motion = [];
   const log = value => lines.push(JSON.stringify(value));
   const assert = (condition, message) => {
     if (!condition) throw new Error(message);
   };
   const readPrefs = async label => {
     const response = await context.request.get('http://localhost:5173/api/prefs');
     const body = await response.text();
     log({ step: label, method: 'GET', url: response.url(),
       status: response.status(), body });
     assert(response.status() === 200, `${label}: GET failed`);
     return JSON.parse(body);
   };

   try {
     log({
       environment: 'http://localhost:5173',
       commit: 'a1b2c3d',
       startedAt: new Date().toISOString(),
       procedure: 'UI click → actual PATCH response → GET → reload; both directions for every notification switch'
     });
     await page.goto('http://localhost:5173/settings/notifications');
     await page.getByRole('heading', { name: 'Notification settings', exact: true }).waitFor();
     const original = await readPrefs('baseline');
     const controls = [
       ['Email notifications', 'emailNotifications'],
       ['Push notifications', 'pushNotifications']
     ];

     for (const [label, key] of controls) {
       const control = page.getByRole('switch', { name: label, exact: true });
       assert(await control.count() === 1, `${label}: ambiguous or missing`);
       assert(
         await control.getAttribute('aria-checked') === String(original[key]),
         `${label}: initial UI differs from GET`
       );

       for (const target of [!original[key], original[key]]) {
         const before = await readPrefs(`${key}: before ${target}`);

         // Observe the actual thumb throughout the click, not a recreated animation.
         await control.evaluate(el => {
           const thumb = el.querySelector('[data-slot="switch-thumb"]');
           if (!thumb) throw new Error('Missing switch thumb');
           window.__motion = { samples: [], durations: [], running: true };
           const sample = () => {
             const data = window.__motion;
             if (!data.running) return;
             data.samples.push({
               at: performance.now(),
               x: thumb.getBoundingClientRect().x
             });
             for (const animation of thumb.getAnimations()) {
               const duration = animation.effect.getTiming().duration;
               if (!data.durations.includes(duration)) data.durations.push(duration);
             }
             requestAnimationFrame(sample);
           };
           sample();
         });

         const patchPromise = page.waitForResponse(response =>
           new URL(response.url()).pathname === '/api/prefs' &&
           response.request().method() === 'PATCH'
         );
         await control.click();
         const patch = await patchPromise;
         log({
           step: `${key}: change to ${target}`,
           method: 'PATCH',
           url: patch.url(),
           requestBody: patch.request().postData(),
           status: patch.status(),
           body: await patch.text()
         });
         assert(patch.status() === 200, `${key}: PATCH not 200`);
         await page.waitForTimeout(250);

         const sampled = await page.evaluate(() => {
           window.__motion.running = false;
           return window.__motion;
         });
         const xs = sampled.samples.map(sample => sample.x);
         const start = xs[0], end = xs[xs.length - 1];
         assert(Math.abs(end - start) > 1, `${key}: thumb did not move`);
         assert(xs.some(x =>
           x > Math.min(start, end) + 0.1 &&
           x < Math.max(start, end) - 0.1
         ), `${key}: no intermediate slide position`);
         assert(sampled.durations.includes(160), `${key}: no 160ms animation`);
         motion.push({ label, target, ...sampled });

         const after = await readPrefs(`${key}: after ${target}`);
         assert(after[key] === target, `${key}: GET did not persist new value`);
         assert(
           JSON.stringify({ ...after, [key]: before[key] }) === JSON.stringify(before),
           `${key}: unrelated preferences changed`
         );
         assert(
           await control.getAttribute('aria-checked') === String(target),
           `${key}: UI state differs from GET`
         );

         await page.reload();
         await page.getByRole('heading', { name: 'Notification settings', exact: true }).waitFor();
         assert(
           await page.getByRole('switch', { name: label, exact: true })
             .getAttribute('aria-checked') === String(target),
           `${key}: reload lost preference`
         );
         log({ step: `${key}: reload`, expected: target, result: 'PASS' });
       }
     }

     // Exercise keyboard activation through the same save path.
     const email = page.getByRole('switch', { name: 'Email notifications', exact: true });
     for (const target of [!original.emailNotifications, original.emailNotifications]) {
       await email.focus();
       const patchPromise = page.waitForResponse(response =>
         new URL(response.url()).pathname === '/api/prefs' &&
         response.request().method() === 'PATCH'
       );
       await page.keyboard.press('Space');
       const patch = await patchPromise;
       log({ step: 'Email notifications: Space', target,
         requestBody: patch.request().postData(), status: patch.status(),
         body: await patch.text() });
       assert(patch.status() === 200, 'Keyboard PATCH failed');
       const prefs = await readPrefs('keyboard follow-up');
       assert(prefs.emailNotifications === target, 'Keyboard save failed');
     }

     const finalPrefs = await readPrefs('restored baseline');
     assert(JSON.stringify(finalPrefs) === JSON.stringify(original), 'Baseline not restored');
     await page.reload();
     await page.getByRole('heading', { name: 'Notification settings', exact: true }).waitFor();
     await page.screenshot({
       path: path.join(art, 'pages/verify/notification-settings.png'),
       animations: 'allow'
     });
     log({ result: 'PASS', capturedAt: new Date().toISOString() });
   } catch (error) {
     log({ result: 'FAIL', error: String(error) });
     throw error;
   } finally {
     await fs.writeFile(path.join(evidence, 'prefs-flow.txt'), lines.join('\n') + '\n');
     await fs.writeFile(path.join(evidence, 'motion-samples.json'), JSON.stringify(motion, null, 2));
     await context.close();
     await page.video().saveAs(path.join(art, 'pages/verify/notification-settings.webm'));
     await fs.rm(path.join(evidence, 'recording-scratch'), { recursive: true, force: true });
     await browser.close();
   }
   ```
