    ```python
    import datetime
    import json
    import subprocess

    BASE = "http://localhost:5173"
    KEYS = ["emailNotifications", "pushNotifications"]

    def request(method, body=None):
        cmd = ["curl", "--silent", "--show-error", "--fail-with-body",
               "--request", method, f"{BASE}/api/prefs",
               "--write-out", "\n%{http_code}"]
        if body is not None:
            cmd += ["--header", "Content-Type: application/json",
                    "--data", json.dumps(body, separators=(",", ":"))]
        print("$ " + subprocess.list2cmdline(cmd), flush=True)
        result = subprocess.run(cmd, text=True, capture_output=True)
        print(result.stdout, flush=True)
        if result.stderr:
            print(result.stderr, flush=True)
        result.check_returncode()
        payload, status = result.stdout.rsplit("\n", 1)
        assert status == "200", status
        return json.loads(payload)

    print("from: local dev http://localhost:5173 @ a1b2c3d")
    print("capturedAt:", datetime.datetime.now(datetime.timezone.utc).isoformat())
    original = request("GET")
    assert all(isinstance(original[k], bool) for k in KEYS)
    try:
        for key in KEYS:
            baseline = request("GET")
            changed = not baseline[key]
            request("PATCH", {key: changed})
            fetched = request("GET")
            assert fetched[key] == changed, (key, fetched)
            assert all(fetched[k] == baseline[k] for k in KEYS if k != key)
            request("PATCH", {key: baseline[key]})
            restored = request("GET")
            assert all(restored[k] == baseline[k] for k in KEYS)
            print(f"PASS {key}: PATCH 200; GET retained changed value; "
                  "other notification preference preserved; reverse change persisted.")
    finally:
        request("PATCH", {k: original[k] for k in KEYS})
        restored = request("GET")
        assert all(restored[k] == original[k] for k in KEYS)
        print("PASS original notification preferences restored.")
    ```
