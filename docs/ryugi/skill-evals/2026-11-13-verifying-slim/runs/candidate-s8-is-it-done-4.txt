   fields = ["id", "title", "content"]
   expected = [
       {field: str(note[field]) for field in fields}
       for note in notes
   ]

   with urlopen(base + "/api/notes/export.csv") as response:
       assert response.status == 200
       assert response.headers.get_content_type() == "text/csv"
       disposition = response.headers.get("Content-Disposition", "")
       assert "attachment" in disposition.lower(), disposition
       assert "notes.csv" in disposition, disposition
       endpoint_bytes = response.read()
       print("GET /api/notes/export.csv: 200")
       print("Content-Type:", response.headers.get("Content-Type"))
       print("Content-Disposition:", disposition)

   def parse(data):
       reader = csv.DictReader(io.StringIO(data.decode("utf-8-sig"), newline=""))
       assert reader.fieldnames == fields, reader.fieldnames
       rows = list(reader)
       assert len(rows) == 42, len(rows)
       assert all(set(row) == set(fields) for row in rows)
       return sorted(rows, key=lambda row: row["id"])

   expected.sort(key=lambda row: row["id"])
   assert parse(endpoint_bytes) == expected
   print("Endpoint CSV: all 42 records and all exported fields match live notes")

   downloaded_bytes = Path("/tmp/notes-export.csv").read_bytes()
   assert parse(downloaded_bytes) == expected
   print("Browser download: all 42 records and all exported fields match live notes")

   contents = [note["content"] for note in expected]
   assert any("," in value for value in contents)
   assert any('"' in value for value in contents)
   assert any("\n" in value for value in contents)
   assert any(any(ord(char) > 127 for char in value) for value in contents)
