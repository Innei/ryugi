7. Shell call to capture API evidence directly, including the command, responses, assertions, restoration, and timestamps:

   ```sh
   cd /Users/p/git/notes-app
   set -o pipefail
   python3 - <<'PY' | tee .ryugi/artifacts/verify/notification-settings/evidence/persistence.txt
   import datetime, json, shlex, subprocess

   def now():
       return datetime.datetime.now(datetime.timezone.utc).isoformat(timespec="seconds")

   def request(method, payload=None):
       command = [
           "curl", "--silent", "--show-error", "--fail-with-body",
           "--request", method, "http://localhost:5173/api/prefs",
           "--write-out", "\nHTTP_STATUS:%{http_code}"
       ]
       if payload is not None:
           command += [
               "--header", "Content-Type: application/json",
               "--data-raw", json.dumps(payload, separators=(",", ":"))
           ]
       print("$ " + shlex.join(command), flush=True)
       result = subprocess.run(command, capture_output=True, text=True)
       print(result.stdout, flush=True)
       if result.stderr:
           print(result.stderr, flush=True)
       result.check_returncode()
       body, status = result.stdout.rsplit("\nHTTP_STATUS:", 1)
       assert status == "200", status
       return json.loads(body) if body.strip() else None

   print("Environment: http://localhost:5173 · local dev @ a1b2c3d")
   print("Started:", now())
   initial = request("GET")
   keys = ("emailNotifications", "pushNotifications")
   assert all(isinstance(initial[k], bool) for k in keys)
   try:
       for key in keys:
           expected = {k: initial[k] for k in keys}
           expected[key] = not initial[key]
           request("PATCH", {key: expected[key]})
           actual = request("GET")
           assert all(actual[k] == expected[k] for k in keys), actual
           print("PASS:", key, "persists; neighbouring preference unchanged")
           request("PATCH", {key: initial[key]})
           restored = request("GET")
           assert all(restored[k] == initial[k] for k in keys), restored
   finally:
       request("PATCH", {k: initial[k] for k in keys})
       restored = request("GET")
       assert all(restored[k] == initial[k] for k in keys), restored
       print("PASS: initial preferences restored")
   print("Finished:", now())
   PY
   ```

   Assumed full output contains every curl command and response: every request returns 200; each following GET contains the changed value and unchanged neighbour; restoration succeeds. Capture interval: `2026-11-13T06:00:00Z`–`2026-11-13T06:00:03Z`.

   ```javascript
   browser.run({
     outputPath: "/Users/p/git/notes-app/.ryugi/artifacts/verify/notification-settings/evidence/ui-flow.txt",
     script: async ({ page, expect, output }) => {
       output("Environment: http://localhost:5173 · local dev @ a1b2c3d");
       output("Started: " + new Date().toISOString());
       await page.reload();

       const names = ["Email notifications", "Push notifications"];
       const keys = ["emailNotifications", "pushNotifications"];
       const initial = await (await page.request.get("/api/prefs")).json();

       for (let i = 0; i < names.length; i++) {
         const toggle = page.getByRole("switch", { name: names[i], exact: true });
         const neighbour = page.getByRole("switch", {
           name: names[1 - i], exact: true
         });
         await expect(toggle).toBeChecked();
         await expect(neighbour).toBeChecked();

         const responsePromise = page.waitForResponse(r =>
           r.url().endsWith("/api/prefs") &&
           r.request().method() === "PATCH"
         );
         await toggle.click();

         const animations = await toggle.evaluate(el =>
           el.getAnimations({ subtree: true }).map(animation => ({
             duration: animation.effect.getTiming().duration,
             keyframes: animation.effect.getKeyframes().map(frame => ({
               transform: frame.transform,
               translate: frame.translate
             }))
           }))
         );
         output(JSON.stringify({ toggle: names[i], animations }));
         const slide = animations.find(animation =>
           animation.duration === 160 &&
           new Set(animation.keyframes.map(frame =>
             JSON.stringify([frame.transform, frame.translate])
           )).size > 1
         );
         expect(slide).toBeTruthy();

         const response = await responsePromise;
         output(names[i] + " UI PATCH status: " + response.status());
         expect(response.status()).toBe(200);
         await expect(toggle).not.toBeChecked();
         await expect(neighbour).toBeChecked();

         const getResponse = await page.request.get("/api/prefs");
         const preferences = await getResponse.json();
         output(JSON.stringify({
           getStatus: getResponse.status(), preferences
         }));
         expect(getResponse.status()).toBe(200);
         expect(preferences[keys[i]]).toBe(false);
         expect(preferences[keys[1 - i]]).toBe(true);

         await page.reload();
         await expect(toggle).not.toBeChecked();
         await expect(neighbour).toBeChecked();
         output("PASS: " + names[i] + " survives reload");

         const restorePromise = page.waitForResponse(r =>
           r.url().endsWith("/api/prefs") &&
           r.request().method() === "PATCH"
         );
         await toggle.focus();
         await page.keyboard.press("Space");
         expect((await restorePromise).status()).toBe(200);
         await expect(toggle).toBeChecked();
         await expect(neighbour).toBeChecked();
         output("PASS: keyboard switching and neighbour unchanged");
         await page.waitForTimeout(250);
       }

       await page.reload();
       const final = await (await page.request.get("/api/prefs")).json();
       for (const key of keys) expect(final[key]).toBe(initial[key]);
       output("PASS: initial preferences restored");
       output("Finished: " + new Date().toISOString());
     }
   })
   ```
