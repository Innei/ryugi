         el => el.getAttribute("aria-label") === name ||

             .map(id => document.getElementById(id)?.textContent || "")

       return el;
     };
     for (const name of names) {
       for (let round = 0; round < 2; round++) {
         const el = find(name);
         const before = el.getAttribute("aria-checked");
         el.click();
         await new Promise(requestAnimationFrame);
         const animations = el.getAnimations({subtree: true});
         const durations = animations.map(a => a.effect.getTiming().duration);
         if (!durations.length || durations.some(d => d !== 160))

         await pause(1000);

         if (before === after) throw new Error(name + ": did not change");
         rows.push({name, before, after, durations});
       }
     }

6. **Shell tool — API persistence, including restoration**

   ```sh
   cd /Users/p/git/notes-app
   python3 - <<'PY' > .ryugi/artifacts/verify/notification-settings/evidence/prefs.txt 2>&1
   import datetime
   import json
   import subprocess

   base = "http://localhost:5173"

   def request(method, body=None):
       command = ["curl", "--silent", "--show-error", "--fail-with-body",
                  "--request", method, "--write-out", "\n%{http_code}"]
       if body is not None:
           command += ["--header", "Content-Type: application/json",
                       "--data", json.dumps(body)]
       command += [base + "/api/prefs"]
       print("$ " + " ".join(command), flush=True)
       result = subprocess.run(command, text=True, capture_output=True)
       print(result.stdout, flush=True)
       if result.stderr:
           print(result.stderr, flush=True)
       result.check_returncode()
       raw, status = result.stdout.rsplit("\n", 1)
       assert status == "200", (method, status)
       return json.loads(raw) if raw.strip() else None

   print("Environment: local dev http://localhost:5173 @ a1b2c3d")
   print("Captured:", datetime.datetime.now(datetime.timezone.utc).isoformat())
   original = request("GET")
   for key in ["emailNotifications", "pushNotifications"]:
       assert isinstance(original[key], bool)
       try:
           changed = not original[key]
           request("PATCH", {key: changed})
           current = request("GET")
           assert current[key] == changed, (key, current)
           assert {k: v for k, v in current.items() if k != key} == {
               k: v for k, v in original.items() if k != key
           }, "Unrelated preferences changed"
           print("PASS:", key, "PATCH 200; GET returned changed value; other preferences preserved")
       finally:
           request("PATCH", {key: original[key]})
           restored = request("GET")
           assert restored == original, "Original preferences were not restored"
           print("PASS:", key, "original preferences restored")
   print("PASS: both notification preferences persist through PATCH then GET")
   PY
   cat .ryugi/artifacts/verify/notification-settings/evidence/prefs.txt
   ```
