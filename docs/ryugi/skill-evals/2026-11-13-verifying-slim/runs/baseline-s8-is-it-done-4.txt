    import csv, io, json
    from datetime import datetime, timezone
    from pathlib import Path
    from urllib.request import urlopen

    base = "http://localhost:5173"
    print("Environment:", base, flush=True)
    print("Captured:", datetime.now(timezone.utc).isoformat(), flush=True)
    print("GET /api/notes", flush=True)
    with urlopen(base + "/api/notes") as response:
        assert response.status == 200
        notes = json.load(response)
    assert len(notes) == 42, len(notes)

    print("GET /api/notes/export.csv", flush=True)
    with urlopen(base + "/api/notes/export.csv") as response:
        print("Status:", response.status, flush=True)
        print("Content-Type:", response.headers.get("Content-Type"), flush=True)
        print("Content-Disposition:", response.headers.get("Content-Disposition"), flush=True)
        assert response.status == 200
        assert response.headers.get_content_type() == "text/csv"
        assert "attachment" in response.headers.get("Content-Disposition", "").lower()
        assert "notes.csv" in response.headers.get("Content-Disposition", "")
        endpoint = response.read()

    fields = ["id", "title", "body"]
    expected = sorted(
        [tuple(str(note[field]) for field in fields) for note in notes]
    )
    for label, data in [
        ("endpoint", endpoint),
        ("browser download", Path(".ryugi/artifacts/verify/csv-export/evidence/browser-download.csv").read_bytes()),
    ]:
        reader = csv.DictReader(io.StringIO(data.decode("utf-8-sig"), newline=""))
        assert reader.fieldnames == fields, reader.fieldnames
        rows = list(reader)
        assert len(rows) == 42, len(rows)
        assert all(set(row) == set(fields) and None not in row.values() for row in rows)
        actual = sorted(tuple(row[field] for field in fields) for row in rows)
        assert actual == expected
        print(label + ": 42 rows; all fields equal the source notes", flush=True)

    bodies = [note["body"] for note in notes]
    for label, predicate in [
        ("comma", lambda value: "," in value),
        ("quote", lambda value: '"' in value),
        ("multiline", lambda value: "\n" in value),
        ("Unicode", lambda value: any(ord(char) > 127 for char in value)),
    ]:
        assert any(predicate(body) for body in bodies), label
        print(label + ": present in fixtures and preserved", flush=True)
    PY
