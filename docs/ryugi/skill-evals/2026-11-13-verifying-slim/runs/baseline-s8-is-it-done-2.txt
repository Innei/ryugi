14. **Write file** `/Users/p/git/notes-app/.ryugi/artifacts/verify/csv-export/evidence/check_csv.py`:
    ```python
    import csv
    import io
    import json
    import urllib.request
    from pathlib import Path

    root = Path(__file__).parent
    base = "http://localhost:5173"

    with urllib.request.urlopen(base + "/api/notes") as response:
        notes = json.load(response)

    with urllib.request.urlopen(base + "/api/notes/export.csv") as response:
        assert response.status == 200
        assert response.headers.get_content_type() == "text/csv"
        assert "attachment" in response.headers["Content-Disposition"].lower()
        endpoint_bytes = response.read()

    downloaded_bytes = (root / "notes.csv").read_bytes()
    assert downloaded_bytes == endpoint_bytes

    reader = csv.DictReader(io.StringIO(downloaded_bytes.decode("utf-8-sig")))
    assert reader.fieldnames == ["id", "title", "body"]
    rows = list(reader)
    assert len(rows) == len(notes) == 42

    def normalized(items):
        return sorted(
            [(str(n["id"]), n["title"], n["body"]) for n in items],
            key=lambda item: item[0],
        )

    assert normalized(rows) == normalized(notes)
    print("PASS: endpoint returns HTTP 200, text/csv and attachment disposition")
    print("PASS: browser download equals endpoint response")
    print("PASS: parsed CSV contains all 42 notes with matching IDs, titles and bodies")
    ```
