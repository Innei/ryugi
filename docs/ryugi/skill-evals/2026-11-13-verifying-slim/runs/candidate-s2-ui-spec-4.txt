8. Browser call to capture and measure motion:

   ```js
   browser.run({
     code: `
       const motionContext = await browser.newContext({
         viewport: { width: 1280, height: 800 },
         reducedMotion: 'no-preference',
         recordVideo: {
           dir: '/Users/p/git/notes-app/.ryugi/artifacts/artboard/settings-redesign/pages/verify',
           size: { width: 1280, height: 800 }
         }
       });
       const motionPage = await motionContext.newPage();
       await motionPage.goto('http://localhost:5173/settings');
       await motionPage.getByRole('switch').first().waitFor();

       for (const label of [
         'Email notifications', 'Push notifications', 'Weekly digest'
       ]) {
         const toggle = motionPage.getByRole('switch', {
           name: label, exact: true
         });
         for (let direction = 0; direction < 2; direction++) {
           const patch = motionPage.waitForResponse(response =>
             new URL(response.url()).pathname === '/api/prefs' &&
             response.request().method() === 'PATCH'
           );
           const timing = await toggle.evaluate(async element => {
             const thumb = element.querySelector('[data-switch-thumb]');
             const before = getComputedStyle(thumb).transform;
             element.click();
             await new Promise(requestAnimationFrame);
             const animations = thumb.getAnimations();
             const slide = animations.find(
               animation => animation.transitionProperty === 'transform'
             );
             if (!slide) throw new Error('No thumb transform transition');
             const duration = slide.effect.getTiming().duration;
             await slide.finished;
             return {
               before,
               after: getComputedStyle(thumb).transform,
               duration
             };
           });
           assert.equal(timing.duration, 160);
           assert.notEqual(timing.before, timing.after);
           assert.equal((await patch).status(), 200);
           console.log(label, direction, timing);
           await motionPage.waitForTimeout(500);
         }
       }
       const video = motionPage.video();
       await motionContext.close();
       await video.saveAs(
         '/Users/p/git/notes-app/.ryugi/artifacts/artboard/settings-redesign/pages/verify/toggle-motion.webm'
       );
       console.log('Recording saved; originals restored by second toggle.');
     `
   })
   ```

   Assumed output: six transform transitions, all with a 160ms duration and different starting and ending transforms. The short recording shows all three controls switching both ways. The browser context uses the same permitted development identity.
