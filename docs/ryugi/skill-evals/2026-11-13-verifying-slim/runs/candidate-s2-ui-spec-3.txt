6. **Browser call — capture and exercise the real app**

   For this simulation, assume the browser tool exposes `browser.run` with Playwright-style `page`, recording controls, and an `outputPath` that saves raw console output. This is an assumed interface, not an available tool invoked here.

   ```javascript
   browser.run({
     viewport: { width: 1280, height: 800 },
     outputPath: "/Users/p/git/notes-app/.ryugi/artifacts/verify/notification-settings/evidence/preferences.txt",
     code: async ({ page, recording }) => {
       const assert = (ok, message) => {
         if (!ok) throw new Error(message);
       };
       const readPrefs = async () => {
         const response = await page.request.get(
           "http://localhost:5173/api/prefs"
         );
         const body = await response.text();
         console.log(`GET /api/prefs ${response.status()}\n${body}`);
         assert(response.status() === 200, "GET must return 200");
         return JSON.parse(body);
       };
       const entries = [
         ["Email notifications", "emailNotifications"],
         ["Push notifications", "pushNotifications"],
         ["Weekly digest", "weeklyDigest"]
       ];

       console.log("Environment: http://localhost:5173, local dev @ a1b2c3d");
       console.log(`Started: ${new Date().toISOString()}`);
       await page.goto("http://localhost:5173/settings");
       const original = await readPrefs();

       for (const [name, key] of entries) {
         const toggle = page.getByRole("switch", { name, exact: true });
         assert(
           (await toggle.getAttribute("aria-checked")) === String(original[key]),
           `${name}: initial UI must match GET`
         );
       }

       await page.screenshot({
         path: "/Users/p/git/notes-app/.ryugi/artifacts/artboard/settings-redesign/pages/verify/notifications.png"
       });
       console.log(`Screenshot captured: ${new Date().toISOString()}`);

       await recording.start({
         path: "/Users/p/git/notes-app/.ryugi/artifacts/artboard/settings-redesign/pages/verify/notifications.webm"
       });

       for (const [name, key] of entries) {
         const toggle = page.getByRole("switch", { name, exact: true });

         for (const expected of [!original[key], original[key]]) {
           const patchPromise = page.waitForResponse(response =>
             new URL(response.url()).pathname === "/api/prefs" &&
             response.request().method() === "PATCH"
           );
           await toggle.click();
           const patch = await patchPromise;
           console.log(
             `PATCH /api/prefs ${patch.status()}\n` +
             `request=${patch.request().postData()}\n` +
             `response=${await patch.text()}`
           );
           assert(patch.status() === 200, "PATCH must return 200");

           const timing = await toggle.locator("[data-switch-thumb]").evaluate(el => {
             const css = getComputedStyle(el);
             return {
               property: css.transitionProperty,
               duration: css.transitionDuration
             };
           });
           console.log(`${name}: ${JSON.stringify(timing)}`);
           assert(
             timing.duration.split(",").every(value => value.trim() === "0.16s"),
             `${name}: thumb transition must be 160ms`
           );

           const persisted = await readPrefs();
           assert(persisted[key] === expected, `${name}: GET must match change`);
           for (const [, otherKey] of entries) {
             if (otherKey !== key) {
               assert(
                 persisted[otherKey] === original[otherKey],
                 `${name}: unrelated preference changed`
               );
             }
           }

           await page.reload();
           assert(
             (await page.getByRole("switch", { name, exact: true })
               .getAttribute("aria-checked")) === String(expected),
             `${name}: reloaded UI must match GET`
           );
           console.log(`${name}: value=${expected}; GET and reload agree`);
         }
       }

       await recording.stop();
       console.log(`Recording captured: ${new Date().toISOString()}`);
       const restored = await readPrefs();
       for (const [, key] of entries) {
         assert(restored[key] === original[key], `${key}: restore failed`);
       }
       console.log("PASS: all three switches exercised both ways; unrelated preferences preserved; original values restored");
     }
   });
   ```
