7. **Shell call — verify the downloaded bytes against the real notes**

   ```sh
   python3 - <<'PY' > .ryugi/artifacts/verify/csv-export/evidence/csv-check.txt 2>&1
   import csv
   import json
   from pathlib import Path

   root = Path(".ryugi/artifacts/verify/csv-export/evidence")
   response = json.loads((root / "export-response.json").read_text())
   assert response["url"] == "http://localhost:5173/api/notes/export.csv"
   assert response["status"] == 200
   headers = {k.lower(): v for k, v in response["headers"].items()}
   assert headers["content-type"].lower().startswith("text/csv")
   assert "attachment" in headers["content-disposition"].lower()
   assert "notes.csv" in headers["content-disposition"]

   notes = json.loads((root / "source-notes.json").read_text())
   with (root / "notes.csv").open(encoding="utf-8-sig", newline="") as stream:
       reader = csv.DictReader(stream)
       assert reader.fieldnames == ["id", "title", "content"], reader.fieldnames
       rows = list(reader)

   expected = {
       str(note["id"]): {key: str(note[key]) for key in ("id", "title", "content")}
       for note in notes
   }
   actual = {row["id"]: row for row in rows}
   assert len(notes) == len(expected) == len(rows) == len(actual) == 42
   assert actual == expected
   values = [note[key] for note in notes for key in ("title", "content")]
   assert any("," in value for value in values)
   assert any('"' in value for value in values)
   assert any("\n" in value for value in values)

   print("Browser-triggered GET /api/notes/export.csv: HTTP 200")
   print("Headers: text/csv; attachment filename notes.csv")
   print("CSV: 42 unique notes; all id/title/content values match the live notes API")
   print("Comma, quote and multiline fixture values round-trip exactly")
   PY
   ```
