8. **Browser call — exercise both directions and measure the actual running animations**

   ```js
   browser.evaluate({
     script: `async () => {
       const results = [];
       const pause = ms => new Promise(resolve => setTimeout(resolve, ms));
       for (const name of ["Email notifications", "Push notifications"]) {
         for (const expected of ["true", "false"]) {
           const control = [...document.querySelectorAll('[role="switch"]')]
             .find(el => el.getAttribute("aria-label") === name);
           if (!control) throw new Error("Missing switch: " + name);
           const thumb = control.querySelector("[data-thumb]");
           if (!thumb) throw new Error("Missing thumb: " + name);
           const before = getComputedStyle(thumb).transform;
           control.click();
           await new Promise(requestAnimationFrame);
           const animations = thumb.getAnimations();
           const slide = animations.find(animation =>
             animation.effect.getKeyframes().some(frame => "transform" in frame));
           if (!slide) throw new Error("Missing transform animation: " + name);
           const duration = slide.effect.getTiming().duration;
           if (duration !== 160) throw new Error("Wrong duration: " + duration);
           await pause(75);
           const middle = getComputedStyle(thumb).transform;
           await pause(200);
           const after = getComputedStyle(thumb).transform;
           if (before === middle || middle === after || before === after)
             throw new Error("Thumb did not slide through an intermediate position");
           if (control.getAttribute("aria-checked") !== expected)
             throw new Error("Wrong switch state");
           results.push({name, checked: expected, duration, before, middle, after});
           await pause(500);
         }
       }
       return {capturedAt: new Date().toISOString(), results};
     }`
   })
   ```

   **Assumed result:** four successful transitions, each `duration: 160`, with distinct start, intermediate, and end transforms. The capture timestamp is `2026-11-13T04:02:00Z`. Both controls finish off.

   These measurements support the screen claim; they do not settle the visual verdict.

    This captures raw output directly into its single evidence location.

    ```sh
    cd /Users/p/git/notes-app
    set -o pipefail
    python3 - <<'PY' 2>&1 | tee .ryugi/artifacts/verify/notification-settings/evidence/persistence.txt
    import datetime
    import json
    import subprocess
    import urllib.request

    base = "http://localhost:5173"
    print("Environment:", base, flush=True)
    print("Commit:", subprocess.check_output(
        ["git", "rev-parse", "--short", "HEAD"], text=True).strip(), flush=True)
    print("Command: python3 urllib.request GET/PATCH /api/prefs; both notification keys, both directions; compare every untouched preference", flush=True)
    print("Started:", datetime.datetime.now(datetime.timezone.utc).isoformat(), flush=True)

    def request(method, payload=None):
        data = None if payload is None else json.dumps(payload).encode()
        req = urllib.request.Request(
            base + "/api/prefs",
            data=data,
            method=method,
            headers={"Content-Type": "application/json"})
        with urllib.request.urlopen(req) as response:
            status = response.status
            raw = response.read().decode()
        print(method, "/api/prefs", "payload=" + json.dumps(payload),
              "status=" + str(status), "body=" + raw, flush=True)
        assert status == 200, (method, status)
        return json.loads(raw) if raw else None

    original = request("GET")
    expected = original.copy()
    try:
        for key in ("emailNotifications", "pushNotifications"):
            for value in (not original[key], original[key]):
                request("PATCH", {key: value})
                expected[key] = value
                actual = request("GET")
                assert actual == expected, {"expected": expected, "actual": actual}
                print("PASS", key, "=", value,
                      "; following GET matches; all other preferences unchanged", flush=True)
        print("PASS original preferences restored", flush=True)
    finally:
        actual = request("GET")
        if actual != original:
            request("PATCH", original)
            assert request("GET") == original
            print("Cleanup restored original preferences", flush=True)
    print("CapturedAt:", datetime.datetime.now(datetime.timezone.utc).isoformat(), flush=True)
    PY
    ```
