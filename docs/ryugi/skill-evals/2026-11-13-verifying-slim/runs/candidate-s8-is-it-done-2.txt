15. **File write** — `/Users/p/git/notes-app/.ryugi/artifacts/verify/csv-export/evidence/check_csv.py`

    ```python
    import csv
    import io
    import json
    from pathlib import Path
    from urllib.request import urlopen

    root = Path(__file__).parent
    base = "http://localhost:5173"

    with urlopen(base + "/api/notes") as response:
        assert response.status == 200
        notes = json.load(response)

    assert len(notes) == 42, len(notes)

    def parse(data):
        reader = csv.DictReader(io.StringIO(data.decode("utf-8-sig"), newline=""))
        assert reader.fieldnames == ["id", "title", "body"], reader.fieldnames
        rows = list(reader)
        assert len(rows) == 42, len(rows)
        assert all(None not in row for row in rows)
        return sorted(
            [(row["id"], row["title"], row["body"]) for row in rows]
        )

    expected = sorted(
        [(str(note["id"]), note["title"], note["body"]) for note in notes]
    )
    assert parse((root / "notes.csv").read_bytes()) == expected
    print("Browser-downloaded CSV: 42 rows; all IDs, titles and bodies match /api/notes.")

    with urlopen(base + "/api/notes/export.csv") as response:
        assert response.status == 200
        assert response.headers.get_content_type() == "text/csv"
        disposition = response.headers.get("Content-Disposition", "")
        assert "attachment" in disposition.lower(), disposition
        assert "notes.csv" in disposition, disposition
        assert parse(response.read()) == expected
        print("GET /api/notes/export.csv: 200; text/csv; attachment notes.csv; 42 matching rows.")

    bodies = [note["title"] + note["body"] for note in notes]
    for label, predicate in [
        ("comma", lambda value: "," in value),
        ("quote", lambda value: '"' in value),
        ("newline", lambda value: "\n" in value),
        ("Unicode", lambda value: any(ord(char) > 127 for char in value)),
    ]:
        assert any(predicate(value) for value in bodies), label
        print(label + " fixture round-trips exactly.")

    print("PASS")
    ```
