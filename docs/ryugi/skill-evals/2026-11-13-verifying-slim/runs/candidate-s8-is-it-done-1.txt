14. **Write file** `.ryugi/artifacts/verify/csv-export/evidence/check_csv.py`:

    ```python
    import csv
    import io
    import json
    from pathlib import Path
    from urllib.request import urlopen

    base = "http://localhost:5173"
    evidence = Path(__file__).parent

    with urlopen(base + "/api/notes") as response:
        assert response.status == 200
        notes = json.load(response)

    assert len(notes) == 42, len(notes)
    expected = {
        str(note["id"]): [str(note["id"]), note["title"], note["content"]]
        for note in notes
    }
    assert len(expected) == 42

    with urlopen(base + "/api/notes/export.csv") as response:
        assert response.status == 200
        assert response.headers.get_content_type() == "text/csv"
        assert response.headers.get_filename() == "notes.csv"
        endpoint_bytes = response.read()

    downloaded_bytes = (evidence / "notes.csv").read_bytes()

    for label, data in [
        ("endpoint", endpoint_bytes),
        ("browser download", downloaded_bytes),
    ]:
        rows = list(csv.reader(
            io.StringIO(data.decode("utf-8-sig"), newline=""),
            strict=True,
        ))
        assert rows[0] == ["id", "title", "content"], rows[0]
        assert len(rows[1:]) == 42, len(rows[1:])
        assert all(len(row) == 3 for row in rows[1:])
        actual = {row[0]: row for row in rows[1:]}
        assert len(actual) == 42
        assert actual == expected
        print(f"PASS {label}: header and all 42 unique records match live notes")

    text = "\n".join(
        value for row in expected.values() for value in row[1:]
    )
    assert "," in text
    assert '"' in text
    assert any("\n" in value for row in expected.values() for value in row[1:])
    assert any(ord(char) > 127 for char in text)
    print("PASS commas, quotes, embedded newlines and Unicode round-trip")
    print("PASS endpoint: HTTP 200, text/csv, attachment filename notes.csv")
    ```
