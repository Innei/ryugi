import csv

import io
import json
import urllib.request
from pathlib import Path

base = "http://localhost:5173"
with urllib.request.urlopen(base + "/api/notes") as response:
    notes = json.load(response)

with urllib.request.urlopen(base + "/api/notes/export.csv") as response:
    assert response.status == 200
    assert response.headers.get_content_type() == "text/csv"
    assert "attachment" in response.headers["Content-Disposition"].lower()
    endpoint_bytes = response.read()

download_bytes = Path(
    ".ryugi/artifacts/verify/csv-export/evidence/download.csv"
).read_bytes()

def parse(raw):
    reader = csv.DictReader(io.StringIO(raw.decode("utf-8-sig"), newline=""))
    assert reader.fieldnames == ["id", "title", "content"], reader.fieldnames
    return list(reader)

expected = [
    {key: str(note[key]) for key in ("id", "title", "content")}
    for note in notes
]
assert len(expected) == 42
assert len({row["id"] for row in expected}) == 42

for label, raw in [("endpoint", endpoint_bytes), ("browser download", download_bytes)]:
    rows = parse(raw)
    assert len(rows) == 42, (label, len(rows))
    assert sorted(rows, key=lambda row: row["id"]) == sorted(
        expected, key=lambda row: row["id"]
