    ```sh
    cd /Users/p/git/notes-app
    python3 - <<'PY' > .ryugi/artifacts/verify/notification-settings/evidence/api-flow.txt 2>&1
    import datetime, json, subprocess

    print("Environment: http://localhost:5173 · local dev @ a1b2c3d", flush=True)
    print("Captured:", datetime.datetime.now(datetime.timezone.utc).isoformat(), flush=True)

    def request(method, payload=None):
        args = ["curl", "--silent", "--show-error", "--fail-with-body",
                "--request", method, "http://localhost:5173/api/prefs",
                "--write-out", "\n%{http_code}"]
        if payload is not None:
            args += ["--header", "Content-Type: application/json",
                     "--data", json.dumps(payload)]
        print("$", subprocess.list2cmdline(args), flush=True)
        result = subprocess.run(args, capture_output=True, text=True)
        print(result.stdout, flush=True)
        print(result.stderr, end="", flush=True)
        assert result.returncode == 0, result.returncode
        body, status = result.stdout.rsplit("\n", 1)
        assert status == "200", status
        return json.loads(body)

    original = request("GET")
    keys = ["emailNotifications", "pushNotifications", "weeklySummary"]
    try:
        for key in keys:
            changed = not original[key]
            request("PATCH", {key: changed})
            actual = request("GET")
            assert actual[key] == changed, (key, actual)
            assert all(actual[k] == original[k] for k in original if k != key)
            request("PATCH", {key: original[key]})
            assert request("GET") == original
            print("PASS:", key, "PATCH 200; following GET returned changed value; neighbors preserved; restored", flush=True)
    finally:
        request("PATCH", {key: original[key] for key in keys})
        assert request("GET") == original
        print("PASS: original preferences restored", flush=True)
    PY
    ```
