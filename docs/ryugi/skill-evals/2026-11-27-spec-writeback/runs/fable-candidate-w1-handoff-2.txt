Records are counted with a CSV parser, because `wc -l` miscounts once quoted fields contain newlines.

**Tool call (shell):**
```
cd /Users/p/git/notes-app && OUT=.artideck/artifacts/verify/export/download-20000.txt && TMP=$(mktemp -d) && {
  echo "\$ git rev-parse --short HEAD"; git rev-parse --short HEAD
  echo "\$ pnpm db:seed --user alice --notes 20000 && pnpm db:seed --user bob --notes 3"
  pnpm db:seed --user alice --notes 20000 | tail -n 1; pnpm db:seed --user bob --notes 3 | tail -n 1
  (pnpm dev > "$TMP/dev.log" 2>&1 &) ; sleep 5
  echo "\$ curl alice /api/notes/export"
  curl -sS -b .dev/alice.cookie -D - -o "$TMP/alice.csv" -w 'http=%{http_code} bytes=%{size_download} ttfb=%{time_starttransfer}s total=%{time_total}s\n' http://localhost:3000/api/notes/export
  echo "\$ parse alice.csv"
  python3 -I -c "import csv,sys; r=list(csv.reader(open(sys.argv[1],encoding='utf-8',newline=''))); print('header=',r[0]); print('records=',len(r)-1); print('unique_ids=',len({x[0] for x in r[1:]})); print('bad_width=',sum(len(x)!=4 for x in r))" "$TMP/alice.csv"
  echo "\$ curl bob /api/notes/export (邻居：小批量 + 权限)"
  curl -sS -b .dev/bob.cookie -o "$TMP/bob.csv" -w 'http=%{http_code}\n' http://localhost:3000/api/notes/export
  python3 -I -c "import csv,sys; a={x[0] for x in list(csv.reader(open(sys.argv[1],encoding='utf-8',newline='')))[1:]}; b=[x[0] for x in list(csv.reader(open(sys.argv[2],encoding='utf-8',newline='')))[1:]]; print('bob_records=',len(b)); print('overlap_with_alice=',len(a&set(b)))" "$TMP/alice.csv" "$TMP/bob.csv"
  echo "\$ curl (no cookie) /api/notes/export"
  curl -sS -o /dev/null -w 'http=%{http_code}\n' http://localhost:3000/api/notes/export
} > "$OUT" 2>&1; pkill -f "pnpm dev"; rm -rf "$TMP"; date -u +%Y-%m-%dT%H:%M:%SZ; cat "$OUT"
```
