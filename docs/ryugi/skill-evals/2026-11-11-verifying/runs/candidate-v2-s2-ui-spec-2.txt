echo "PATCH status: $CODE"
echo '$ curl -s GET /api/prefs   (after)'
AFTER=$(curl -s $B); echo "$AFTER"
GOT=$(echo "$AFTER" | jq -r .emailDigest)
echo "expected emailDigest=$NEW got=$GOT"
[ "$CODE" = 200 ] && [ "$GOT" = "$NEW" ] && echo "RESULT: PASS" || echo "RESULT: FAIL"
echo "# restoring original value"
curl -s -o /dev/null -w "restore PATCH status: %{http_code}\n" -X PATCH -H 'Content-Type: application/json' -d "{\"emailDigest\":$OLD}" $B

cat $OUT
