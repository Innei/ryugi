```sh
name: Release

on: push

  release:
    runs-on: macos-latest
    steps:
      - uses: actions/checkout@v4
      - name: Build and sign
        env:
          SIGNING_IDENTITY: ${{ secrets.SIGNING_IDENTITY }}

          else
            echo "workflow: SIGNING_IDENTITY empty or unset"
          fi
          ./scripts/build.sh
EOF
cat > scripts/build.sh <<'EOF'
#!/bin/bash
set -euo pipefail

if [ -n "${SIGNING_IDENTITY:-}" ]; then
  echo "build: SIGNING_IDENTITY set"
else
  echo "build: SIGNING_IDENTITY empty or unset"
fi
export IDENTITY="$SIGNING_IDENTITY"
if [ -n "$IDENTITY" ]; then
  echo "build: IDENTITY exported and nonempty"
else
  echo "build: IDENTITY exported but empty"
fi
xcodebuild -project MacApp.xcodeproj -scheme MacApp -configuration Release CONFIGURATION_BUILD_DIR="$PWD/build"
./scripts/sign.sh
EOF
cat > scripts/sign.sh <<'EOF'
#!/bin/bash
set -euo pipefail

KEYCHAIN="$RUNNER_TEMP/signing.keychain-db"
CERTIFICATE="$RUNNER_TEMP/signing.p12"

cleanup() {
  rm -f "$CERTIFICATE"
  security delete-keychain "$KEYCHAIN"
}
trap cleanup EXIT

if [ -n "${IDENTITY:-}" ]; then
  echo "sign: IDENTITY set"
else
  echo "sign: IDENTITY empty or unset"
fi

printf '%s' "$CERTIFICATE_BASE64" | base64 --decode > "$CERTIFICATE"
security create-keychain -p "$KEYCHAIN_PASSWORD" "$KEYCHAIN"
security unlock-keychain -p "$KEYCHAIN_PASSWORD" "$KEYCHAIN"
security import "$CERTIFICATE" -k "$KEYCHAIN" -P "$CERTIFICATE_PASSWORD" -T /usr/bin/codesign
security set-key-partition-list -S apple-tool:,apple:,codesign: -s -k "$KEYCHAIN_PASSWORD" "$KEYCHAIN"

echo "sign: user keychain search list"
security list-keychains -d user
echo "sign: identities in temporary keychain"
security find-identity -v -p codesigning "$KEYCHAIN"
echo "sign: identities in default search list"
security find-identity -v -p codesigning
echo "sign: requested identity present in temporary keychain"
if security find-certificate -c "$IDENTITY" "$KEYCHAIN" >/dev/null 2>&1; then
  echo "sign: requested certificate found"
else
  echo "sign: requested certificate absent or lookup failed"
fi

codesign --force --options runtime --sign "$IDENTITY" build/MacApp.app
EOF
```
