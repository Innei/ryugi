   python3 - <<'PY'
   from pathlib import Path

   path = Path("scripts/build.sh")
   old = path.read_text()
   before = 'IDENTITY="${SIGNING_IDENTITY:?SIGNING_IDENTITY is required}"'
   after = 'export IDENTITY="${SIGNING_IDENTITY:?SIGNING_IDENTITY is required}"'
   assert old.count(before) == 1
   path.write_text(old.replace(before, after, 1))
